     1→/**
     2→ * POST /api/bot/send
     3→ * Body: { tenantSlug, phone, message?: text, button?: id, location?: {lat,lng}, mediaType?: "voice"|"image" }

     6→ *
     7→ * BotReply = { kind: "text" | "buttons" | "list", text, buttons?, sections? }

     9→ * State machine: language → menu → service_draft → address → confirm → done
    10→ * Resilient: "cancel"/"menu" anywhere → back to menu. Old button taps → re-show current step.
    11→ */

   110→function buildMenuList(tenantId: string, services: Array<{ id: string; key: string; kind: string; icon: string; labels: string }>, lang: "en" | "hi") {
   111→  const rows = services
   112→    .filter((s) => s.kind !== "custom" || s.key === "custom")
   113→    .sort((a, b) => {
   114→      const order = ["cake", "grocery", "chicken", "parcel", "ride", "repair", "team", "custom"];
   115→      return order.indexOf(a.key) - order.indexOf(b.key);
   116→    })
   117→    .map((s) => {
   118→      const labels = safeParse<Record<string, string>>(s.labels, {});
   119→      return {
   120→        id: `svc_${s.id}`,
   121→        title: `${s.icon} ${labels[lang] || labels.en || s.key}`,
   122→        description: s.kind === "order" ? "Order now" : s.kind === "book" ? "Book a slot" : s.kind === "team" ? "Talk to us" : "Tell us what you need",
   123→      };
   124→    });
   125→  return rows;
   126→}

   132→export async function POST(req: NextRequest) {
   133→  const body = await req.json();
   134→  const { tenantSlug, phone, message, button, location, mediaType } = body as {
   135→    tenantSlug: string;
   136→    phone: string;
   137→    message?: string;
   138→    button?: string;
   139→    location?: { lat: number; lng: number };
   140→    mediaType?: "voice" | "image";
   141→  };
   142→
   143→  if (!tenantSlug || !phone) {
   144→    return NextResponse.json({ error: "tenantSlug and phone required" }, { status: 400 });
   145→  }
   146→
   147→  const tenant = await db.tenant.findUnique({
   148→    where: { slug: tenantSlug },
   149→    include: {
   150→      services: { where: { isActive: true } },
   151→      cities: { where: { isActive: true } },
   152→    },
   153→  });
   154→  if (!tenant) return NextResponse.json({ error: "tenant not found" }, { status: 404 });
   155→
   156→  // Find or create customer
   157→  let customer = await db.customer.findUnique({
   158→    where: { tenantId_phone: { tenantId: tenant.id, phone } },
   159→  });
   160→  if (!customer) {
   161→    customer = await db.customer.create({
   162→      data: { tenantId: tenant.id, phone, language: "en" },
   163→    });
   164→  }
   165→
   166→  // Find or create session
   167→  let session = await db.botSession.findUnique({
   168→    where: { tenantId_phone: { tenantId: tenant.id, phone } },
   169→  });
   170→  if (!session) {
   171→    session = await db.botSession.create({
   172→      data: { tenantId: tenant.id, customerId: customer.id, phone },
   173→    });
   174→  }
   175→
   176→  const lang = (customer.language as "en" | "hi") || "en";
   177→  const replies: BotReply[] = [];
   178→
   179→  // ── Handle charges agree/cancel buttons (sent by provider via /charges) ──
   180→  if (button && button.startsWith("charges_agree_")) {
   181→    const orderId = button.slice("charges_agree_".length);
   182→    try {
   183→      await fetch(`${req.nextUrl.origin}/api/orders/${orderId}/charges`, {
   184→        method: "PATCH",
   185→        headers: { "Content-Type": "application/json" },
   186→        body: JSON.stringify({ agreed: true, tenantSlug }),
   187→      });
   188→      replies.push({ kind: "text", text: "✅ Charges agreed. The provider will send a payment link shortly." });
   189→    } catch {
   190→      replies.push({ kind: "text", text: "Something went wrong. Please try again." });
   191→    }
   192→    return NextResponse.json({ replies, session: { state: session.state } });
   193→  }
   194→  if (button && button.startsWith("charges_cancel_")) {
   195→    const orderId = button.slice("charges_cancel_".length);
   196→    try {
   197→      await fetch(`${req.nextUrl.origin}/api/orders/${orderId}/charges`, {
   198→        method: "PATCH",
   199→        headers: { "Content-Type": "application/json" },
   200→        body: JSON.stringify({ agreed: false, tenantSlug }),
   201→      });
   202→      replies.push({ kind: "text", text: "❌ Order cancelled. Type *menu* to start a new order." });
   203→    } catch {
   204→      replies.push({ kind: "text", text: "Something went wrong." });
   205→    }
   206→    return NextResponse.json({ replies, session: { state: session.state } });
   207→  }
   208→
   209→  // ── Handle rating buttons (rate_5_<orderId>, rate_4_, rate_3_) ──
   210→  if (button && button.startsWith("rate_")) {
   211→    const match = button.match(/^rate_(\d+)_(.+)$/);
   212→    if (match) {
   213→      const rating = parseInt(match[1], 10);
   214→      const orderId = match[2];
   215→      try {
   216→        const res = await fetch(`${req.nextUrl.origin}/api/orders/${orderId}/review`, {
   217→          method: "POST",
   218→          headers: { "Content-Type": "application/json" },
   219→          body: JSON.stringify({ rating, tenantSlug }),
   220→        });
   221→        const data = await res.json();
   222→        if (res.ok) {
   223→          replies.push({ kind: "text", text: `🙏 Thank you for your ${rating}★ rating!${data.googleReviewUrl ? "\n\nWe've also sent you a link to review us on Google — it really helps!" : ""}` });
   224→        } else if (data.error === "review_already_submitted") {
   225→          replies.push({ kind: "text", text: "You've already rated this order. Thank you! 🙏" });
   226→        } else {
   227→          replies.push({ kind: "text", text: "Thanks for your feedback! 🙏" });
   228→        }
   229→      } catch {
   230→        replies.push({ kind: "text", text: "Thanks for your feedback! 🙏" });
   231→      }
   232→      return NextResponse.json({ replies, session: { state: session.state } });
   233→    }
   234→  }
   235→
   236→  // ── Handle "cancel" / "menu" anywhere ────────────────────
   237→  const text = (message || "").trim().toLowerCase();
   238→  if (text === "cancel" || text === "❌ cancel" || text === "cancel order") {
   239→    // Reset draft
   240→    await db.botSession.update({
   241→      where: { id: session.id },
   242→      data: {
   243→        state: "menu",
   244→        draftService: null,
   245→        draftItems: "[]",
   246→        draftTiming: null,
   247→        draftShop: null,
   248→        draftAddress: null,
   249→        draftLat: null,
   250→        draftLng: null,
   251→      },
   252→    });
   253→    replies.push({ kind: "text", text: t(tenant, lang, "cancelled") });
   254→    // Show menu
   255→    const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   256→    replies.push({
   257→      kind: "list",
   258→      text: t(tenant, lang, "menu_title"),
   259→      listTitle: t(tenant, lang, "menu_title"),
   260→      listButton: t(tenant, lang, "menu_button"),
   261→      sections: [{ title: "Services", rows: menuRows }],
   262→    });
   263→    return NextResponse.json({ replies, session: { state: "menu" } });
   264→  }
   265→
   266→  if (text === "menu" || text === "start" || text === "hi" || text === "hello") {
   267→    await db.botSession.update({
   268→      where: { id: session.id },
   269→      data: { state: "menu", draftService: null, draftItems: "[]" },
   270→    });
   271→    const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   272→    replies.push({
   273→      kind: "list",
   274→      text: t(tenant, lang, "menu_title"),
   275→      listTitle: t(tenant, lang, "menu_title"),
   276→      listButton: t(tenant, lang, "menu_button"),
   277→      sections: [{ title: "Services", rows: menuRows }],
   278→    });
   279→    return NextResponse.json({ replies, session: { state: "menu" } });
   280→  }
   281→
   282→  // ── State machine ────────────────────────────────────────
   283→  switch (session.state) {
   284→    case "language": {
   285→      if (button === "lang_en") {
   286→        await db.customer.update({ where: { id: customer.id }, data: { language: "en" } });
   287→        const menuRows = buildMenuList(tenant.id, tenant.services, "en");
   288→        await db.botSession.update({ where: { id: session.id }, data: { state: "menu" } });
   289→        replies.push({
   290→          kind: "list",
   291→          text: t(tenant, "en", "menu_title"),
   292→          listTitle: t(tenant, "en", "menu_title"),
   293→          listButton: t(tenant, "en", "menu_button"),
   294→          sections: [{ title: "Services", rows: menuRows }],
   295→        });
   296→      } else if (button === "lang_hi") {
   297→        await db.customer.update({ where: { id: customer.id }, data: { language: "hi" } });
   298→        const menuRows = buildMenuList(tenant.id, tenant.services, "hi");
   299→        await db.botSession.update({ where: { id: session.id }, data: { state: "menu" } });
   300→        replies.push({
   301→          kind: "list",
   302→          text: t(tenant, "hi", "menu_title"),
   303→          listTitle: t(tenant, "hi", "menu_title"),
   304→          listButton: t(tenant, "hi", "menu_button"),
   305→          sections: [{ title: "Services", rows: menuRows }],
   306→        });
   307→      } else {
   308→        replies.push({
   309→          kind: "buttons",
   310→          text: t(tenant, lang, "welcome"),
   311→          buttons: [
   312→            { id: "lang_en", label: "🇬🇧 English" },
   313→            { id: "lang_hi", label: "🇮🇳 हिन्दी" },
   314→          ],
   315→        });
   316→      }
   317→      break;
   318→    }
   319→
   320→    case "menu": {
   321→      // Handle list item tap: svc_<id>
   322→      if (button && button.startsWith("svc_")) {
   323→        const svcId = button.slice(4);
   324→        const svc = tenant.services.find((s) => s.id === svcId);
   325→        if (!svc) {
   326→          replies.push({ kind: "text", text: t(tenant, lang, "invalid") });
   327→          break;
   328→        }
   329→        await db.botSession.update({
   330→          where: { id: session.id },
   331→          data: { state: "service_draft", draftService: svc.id, draftItems: "[]", draftShop: null, draftTiming: null },
   332→        });
   333→        // Service-specific first message
   334→        if (svc.key === "grocery") {
   335→          replies.push({ kind: "buttons", text: t(tenant, lang, "ask_grocery"), buttons: [
   336→            { id: "grocery_done", label: "✅ Done" },
   337→            { id: "grocery_more", label: "➕ Add more" },
   338→            { id: "cancel", label: "❌ Start over" },
   339→          ]});
   340→        } else if (svc.kind === "order") {
   341→          replies.push({ kind: "text", text: t(tenant, lang, `ask_${svc.key}`) });
   342→        } else if (svc.kind === "book") {
   343→          replies.push({ kind: "text", text: t(tenant, lang, `ask_${svc.key}`) });
   344→        } else if (svc.kind === "team") {
   345→          replies.push({ kind: "text", text: t(tenant, lang, "ask_team") });
   346→          // Mark as custom
   347→          await db.botSession.update({
   348→            where: { id: session.id },
   349→            data: { state: "service_draft", draftService: svc.id },
   350→          });
   351→        } else if (svc.kind === "custom") {
   352→          replies.push({ kind: "text", text: t(tenant, lang, "ask_custom") });
   353→        }
   354→      } else {
   355→        // Re-show menu
   356→        const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   357→        replies.push({
   358→          kind: "list",
   359→          text: t(tenant, lang, "old_button") + "\n" + t(tenant, lang, "menu_title"),
   360→          listTitle: t(tenant, lang, "menu_title"),
   361→          listButton: t(tenant, lang, "menu_button"),
   362→          sections: [{ title: "Services", rows: menuRows }],
   363→        });
   364→      }
   365→      break;
   366→    }
   367→
   368→    case "service_draft": {
   369→      const svcId = session.draftService;
   370→      if (!svcId) {
   371→        // Reset
   372→        await db.botSession.update({ where: { id: session.id }, data: { state: "menu" } });
   373→        replies.push({ kind: "text", text: t(tenant, lang, "back_to_menu") });
   374→        break;
   375→      }
   376→      const svc = tenant.services.find((s) => s.id === svcId);
   377→      if (!svc) {
   378→        await db.botSession.update({ where: { id: session.id }, data: { state: "menu", draftService: null } });
   379→        replies.push({ kind: "text", text: t(tenant, lang, "back_to_menu") });
   380→        break;
   381→      }
   382→
   383→      // ── Grocery: accumulate items until Done ──────────
   384→      if (svc.key === "grocery") {
   385→        if (button === "grocery_done") {
   386→          const items = safeParse<{ name: string; qty?: string | number }[]>(session.draftItems, []);
   387→          if (items.length === 0) {
   388→            replies.push({ kind: "text", text: "Your list is empty. Add items first." });
   389→            break;
   390→          }
   391→          replies.push({ kind: "text", text: t(tenant, lang, "grocery_done") });
   392→          // Move to shop question
   393→          await db.botSession.update({
   394→            where: { id: session.id },
   395→            data: { state: "shop" },
   396→          });
   397→          replies.push({
   398→            kind: "buttons",
   399→            text: t(tenant, lang, "ask_shop"),
   400→            buttons: [
   401→              { id: "shop_any", label: "🏪 Any shop" },
   402→              { id: "shop_specify", label: "✏️ Specify" },
   403→            ],
   404→          });
   405→          break;
   406→        }
   407→        if (button === "grocery_more" || message || mediaType) {
   408→          const items = safeParse<{ name: string; qty?: string | number }[]>(session.draftItems, []);
   409→          // ── Voice note: transcribe via AI, then extract items ──
   410→          if (mediaType === "voice") {
   411→            replies.push({ kind: "text", text: t(tenant, lang, "voice_received") });

   883→            description: items[0]?.name || null,
   884→            preferredShop: session.draftShop,
   885→            timing: session.draftTiming,
   886→            addressText: session.draftAddress,
   887→            addressArea: session.draftAddress ? session.draftAddress.split(" (")[0] : null,
   888→            addressLat: session.draftLat,
   889→            addressLng: session.draftLng,
   890→            source: "bot",
   891→            activity: {
   892→              create: [
   893→                { tenantId: tenant.id, actor: "bot", action: "created", detail: "Order created via WhatsApp" },
   894→              ],
   895→            },
   896→          },
   897→        });
   898→        // Broadcast to matching providers
   899→        const matchingProviders = await db.provider.findMany({
   900→          where: {
   901→            tenantId: tenant.id,
   902→            cityId,
   903→            isOnline: true,
   904→            isActive: true,
   905→          },
   906→        });
   907→        const toBroadcast = svc
   908→          ? matchingProviders.filter((p) => {
   909→              const sids = safeParse<string[]>(p.serviceIds, []);
   910→              return sids.length === 0 || sids.includes(svc.id);
   911→            })
   912→          : matchingProviders;
   913→
   914→        if (toBroadcast.length > 0) {
   915→          await db.order.update({ where: { id: order.id }, data: { status: "broadcast" } });
   916→          await db.orderBroadcast.createMany({
   917→            data: toBroadcast.map((p) => ({ orderId: order.id, providerId: p.id, status: "pending" })),
   918→          });
   919→          await db.activity.create({
   920→            data: {
   921→              tenantId: tenant.id, orderId: order.id,
   922→              actor: "system", action: "broadcast",
   923→              detail: `Broadcast to ${toBroadcast.length} provider(s)`,
   924→            },
   925→          });
   926→        }
   927→
   928→        await db.botSession.update({
   929→          where: { id: session.id },
   930→          data: {
   931→            state: "done",
   932→            draftService: null,
   933→            draftItems: "[]",
   934→            draftShop: null,
   935→            draftTiming: null,
   936→            draftAddress: null,
   937→            draftLat: null,
   938→            draftLng: null,
   939→          },
   940→        });
   941→        replies.push({ kind: "text", text: t(tenant, lang, "placed", { code }) });
   942→        // Show menu again
   943→        const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   944→        replies.push({
   945→          kind: "list",
   946→          text: t(tenant, lang, "menu_title"),
   947→          listTitle: t(tenant, lang, "menu_title"),
   948→          listButton: t(tenant, lang, "menu_button"),
   949→          sections: [{ title: "Services", rows: menuRows }],
   950→        });
   951→        return NextResponse.json({ replies, session: { state: "done" }, order });
   952→      }
   953→      if (button === "confirm_change") {
   954→        // Restart from service
   955→        await db.botSession.update({
   956→          where: { id: session.id },
   957→          data: { state: "service_draft", draftItems: "[]" },
   958→        });
   959→        replies.push({ kind: "text", text: "Let's start over. Tell us what you need:" });
   960→        break;
   961→      }
   962→      if (button === "cancel") {
   963→        await db.botSession.update({
   964→          where: { id: session.id },
   965→          data: { state: "menu", draftService: null, draftItems: "[]" },
   966→        });
   967→        replies.push({ kind: "text", text: t(tenant, lang, "cancelled") });
   968→        const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   969→        replies.push({
   970→          kind: "list",
   971→          text: t(tenant, lang, "menu_title"),
   972→          listTitle: t(tenant, lang, "menu_title"),
   973→          listButton: t(tenant, lang, "menu_button"),
   974→          sections: [{ title: "Services", rows: menuRows }],
   975→        });
   976→      }
   977→      break;
   978→    }
   979→
   980→    case "done":
   981→    default: {
   982→      // Show menu
   983→      const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   984→      await db.botSession.update({ where: { id: session.id }, data: { state: "menu" } });
   985→      replies.push({
   986→        kind: "list",
   987→        text: t(tenant, lang, "menu_title"),
   988→        listTitle: t(tenant, lang, "menu_title"),
   989→        listButton: t(tenant, lang, "menu_button"),
   990→        sections: [{ title: "Services", rows: menuRows }],
   991→      });
   992→      break;
   993→    }
   994→  }
   995→
   996→  // Update session lastMessageAt
   997→  await db.botSession.update({
   998→    where: { id: session.id },
   999→    data: { lastMessageAt: new Date() },
  1000→  });
  1001→
  1002→  return NextResponse.json({ replies, session: { state: session.state } });
  1003→}

  1005→function buildOrderSummary(
  1006→  svc: { icon: string; key: string; labels: string } | undefined,
  1007→  items: Array<{ name: string; qty?: string | number }>,
  1008→  shop: string | null,
  1009→  timing: string | null,
  1010→  address: string,
  1011→  area: string | null,
  1012→  lang: "en" | "hi"
  1013→): string {
  1014→  const labels = svc ? safeParse<Record<string, string>>(svc.labels, {}) : {};
  1015→  const svcName = svc ? `${svc.icon} ${labels[lang] || labels.en || svc.key}` : "Order";
  1016→  const lines = [
  1017→    `*${svcName}*`,
  1018→    "",
  1019→    "🛍️ Items:",
  1020→    ...items.map((it, i) => `  ${i + 1}. ${it.name}${it.qty ? ` ×${it.qty}` : ""}`),
  1021→  ];
  1022→  if (shop) lines.push("", `🏪 Shop: ${shop}`);
  1023→  if (timing) lines.push(`⏰ Timing: ${timing}`);
  1024→  lines.push("", `📍 Address:`, `  ${address}`);
  1025→  if (area) lines.push(`  Area: ${area}`);
  1026→  return lines.join("\n");
  1027→}
