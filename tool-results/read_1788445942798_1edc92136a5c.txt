   110→   110→/** Merge grocery items by normalized name — sums quantities for duplicates (H9 fix) */
   111→   111→function mergeItems(existing: Array<{ name: string; qty?: string | number }>, newItems: Array<{ name: string; qty?: string | number }>): Array<{ name: string; qty?: string | number }> {
   112→   112→  const map = new Map<string, { name: string; qty?: string | number }>();
   113→   113→  for (const it of [...existing, ...newItems]) {
   114→   114→    const key = it.name.toLowerCase().trim();
   115→   115→    const existingEntry = map.get(key);
   116→   116→    if (existingEntry) {
   117→   117→      // Try to sum quantities if both are numeric
   118→   118→      const existingQty = parseFloat(String(existingEntry.qty || "0"));
   119→   119→      const newQty = parseFloat(String(it.qty || "0"));
   120→   120→      if (!isNaN(existingQty) && !isNaN(newQty)) {
   121→   121→        existingEntry.qty = existingQty + newQty;
   122→   122→      } else {
   123→   123→        // Can't sum — keep the newer one
   124→   124→        existingEntry.qty = it.qty || existingEntry.qty;
   125→   125→      }
   126→   126→    } else {
   127→   127→      map.set(key, { ...it });
   128→   128→    }
   129→   129→  }
   130→   130→  return Array.from(map.values());
   131→   131→}

  1056→  1056→function buildOrderSummary(
  1057→  1057→  svc: { icon: string; key: string; labels: string } | undefined,
  1058→  1058→  items: Array<{ name: string; qty?: string | number }>,
  1059→  1059→  shop: string | null,
  1060→  1060→  timing: string | null,
  1061→  1061→  address: string,
  1062→  1062→  area: string | null,
  1063→  1063→  lang: "en" | "hi"
  1064→  1064→): string {
  1065→  1065→  const labels = svc ? safeParse<Record<string, string>>(svc.labels, {}) : {};
  1066→  1066→  const svcName = svc ? `${svc.icon} ${labels[lang] || labels.en || svc.key}` : "Order";
  1067→  1067→  const lines = [
  1068→  1068→    `*${svcName}*`,
  1069→  1069→    "",
  1070→  1070→    "🛍️ Items:",
  1071→  1071→    ...items.map((it, i) => `  ${i + 1}. ${it.name}${it.qty ? ` ×${it.qty}` : ""}`),
  1072→  1072→  ];
  1073→  1073→  if (shop) lines.push("", `🏪 Shop: ${shop}`);
  1074→  1074→  if (timing) lines.push(`⏰ Timing: ${timing}`);
  1075→  1075→  lines.push("", `📍 Address:`, `  ${address}`);
  1076→  1076→  if (area) lines.push(`  Area: ${area}`);
  1077→  1077→  return lines.join("\n");
  1078→  1078→}
