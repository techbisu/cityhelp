    49→export function ProviderApp() {
    50→  const setView = useApp((s) => s.setView);
    51→  const providerId = useApp((s) => s.providerId);
    52→  const providerTenantSlug = useApp((s) => s.providerTenantSlug);
    53→  const setProvider = useApp((s) => s.setProvider);
    54→  const clearProvider = useApp((s) => s.clearProvider);
    55→
    56→  const [provider, setProviderInfo] = useState<ProviderInfo | null>(null);
    57→  const [activeJobs, setActiveJobs] = useState<Job[]>([]);
    58→  const [pastJobs, setPastJobs] = useState<Job[]>([]);
    59→  const [customRequests, setCustomRequests] = useState<Job[]>([]);
    60→  const [incomingJob, setIncomingJob] = useState<Job | null>(null);
    61→  const [view, setLocalView] = useState<"home" | "job" | "custom" | "new" | "history" | "onboard">("home");
    62→  const [selectedJob, setSelectedJob] = useState<Job | null>(null);
    63→
    64→  // Login form state
    65→  const [loginPhone, setLoginPhone] = useState("+919811100001");
    66→  const [loginPin, setLoginPin] = useState("1234");
    67→  const [loginError, setLoginError] = useState<string | null>(null);
    68→  const [loginLoading, setLoginLoading] = useState(false);
    69→
    70→  // ── If logged in, fetch provider info ─────────────────
    71→  const refresh = useCallback(async () => {
    72→    if (!providerId) return;
    73→    const res = await fetch(`/api/providers/${providerId}?jobs=true`);
    74→    if (!res.ok) {
    75→      clearProvider();
    76→      return;
    77→    }
    78→    const data = await res.json();
    79→    // Map tenant relation fields to flat fields expected by the provider state
    80→    const p = data.provider;
    81→    const tenantData = p.tenant || {};
    82→    setProviderInfo({
    83→      ...p,
    84→      tenantSlug: tenantData.slug || p.tenantSlug,
    85→      tenantName: tenantData.name || p.tenantName,
    86→      tenantAccent: tenantData.accentColor || p.tenantAccent,
    87→      cityName: p.city?.name || p.cityName,
    88→    });
    89→    setActiveJobs(data.activeJobs);
    90→    setPastJobs(data.pastJobs);
    91→    setCustomRequests(data.customRequests);
    92→  }, [providerId, clearProvider]);
    93→
    94→  useEffect(() => {
    95→    if (providerId) {
    96→      refresh();
    97→      const id = setInterval(refresh, 4000);
    98→      return () => clearInterval(id);
    99→    }
   100→  }, [providerId, refresh]);
   101→
   102→  // ── Poll for incoming broadcast jobs ──────────────────
   103→  useEffect(() => {
   104→    if (!provider || !provider.isOnline || !provider.tenantSlug) return;
   105→    let cancelled = false;
   106→    async function checkIncoming() {
   107→      if (cancelled || !provider || !provider.tenantSlug) return;
   108→      try {
   109→        const res = await fetch(`/api/orders?tenantSlug=${provider.tenantSlug}&status=broadcast`);
   110→        const data = await res.json();
   111→        const myBroadcasts = (data.orders as Job[]).filter((o) => {
   112→          // Only show orders in my city, with my service
   113→          if (o.city.name !== provider.cityName) return false;
   114→          // Check if this provider is in the broadcast list — for demo, accept all broadcast orders in my city
   115→          return true;
   116→        });
   117→        if (myBroadcasts.length > 0 && !incomingJob) {
   118→          // Pick the newest
   119→          setIncomingJob(myBroadcasts[0]);
   120→        }
   121→      } catch {
   122→        // silent
   123→      }
   124→    }
   125→    checkIncoming();
   126→    const id = setInterval(checkIncoming, 5000);
   127→    return () => { cancelled = true; clearInterval(id); };
   128→  }, [provider, incomingJob]);
   129→
   130→  // ── Login ─────────────────────────────────────────────
   131→  async function handleLogin() {
   132→    setLoginLoading(true);
   133→    setLoginError(null);
   134→    try {
   135→      const res = await fetch("/api/providers/login", {
   136→        method: "POST",
   137→        headers: { "Content-Type": "application/json" },
   138→        body: JSON.stringify({ tenantSlug: providerTenantSlug || "shanti", phone: loginPhone, pin: loginPin }),
   139→      });
   140→      const data = await res.json();
   141→      if (!res.ok) {
   142→        setLoginError(data.message || data.error || "Login failed");
   143→        return;
   144→      }
   145→      setProvider(data.provider.id, data.provider.tenantSlug);
   146→      setProviderInfo(data.provider);
   147→      setLocalView("onboard");
   148→    } finally {
   149→      setLoginLoading(false);
   150→    }
   151→  }
   152→
   153→  async function toggleOnline() {
   154→    if (!provider) return;
   155→    const newOnline = !provider.isOnline;
   156→    await fetch("/api/providers", {
   157→      method: "PATCH",
   158→      headers: { "Content-Type": "application/json" },
   159→      body: JSON.stringify({ id: provider.id, isOnline: newOnline }),
   160→    });
   161→    setProviderInfo({ ...provider, isOnline: newOnline });
   162→    if (newOnline) toast.success("You're now online — ready to receive jobs");
   163→    else toast.info("You're now offline");
   164→  }
   165→
   166→  async function handleAccept() {
   167→    if (!incomingJob || !provider) return;
   168→    const res = await fetch(`/api/orders/${incomingJob.id}/accept`, {
   169→      method: "POST",
   170→      headers: { "Content-Type": "application/json" },
   171→      body: JSON.stringify({ providerId: provider.id }),
   172→    });
   173→    const data = await res.json();
   174→    if (res.status === 409) {
   175→      toast.error("Already taken by another provider");
   176→      setIncomingJob(null);
   177→      return;
   178→    }
   179→    toast.success(`✅ Order #${incomingJob.code} accepted!`);
   180→    setIncomingJob(null);
   181→    refresh();
   182→  }
   183→
   184→  function handleReject() {
   185→    if (!incomingJob || !provider) return;
   186→    fetch(`/api/orders/${incomingJob.id}/reject`, {
   187→      method: "POST",
   188→      headers: { "Content-Type": "application/json" },
   189→      body: JSON.stringify({ providerId: provider.id }),
   190→    });
   191→    toast.info("Rejected — waiting for next job");
   192→    setIncomingJob(null);
   193→  }
   194→
   195→  async function updateJobStatus(job: Job, status: "picked" | "delivered") {
   196→    await fetch(`/api/orders/${job.id}/status`, {
   197→      method: "POST",
   198→      headers: { "Content-Type": "application/json" },
   199→      body: JSON.stringify({ status, providerId: provider?.id }),
   200→    });
   201→    toast.success(status === "picked" ? "📦 Marked as picked up" : "✅ Marked as delivered");
   202→    refresh();
   203→    setSelectedJob(null);
   204→    setLocalView("home");
   205→  }
   206→
   207→  // ── Render: login screen ──────────────────────────────
   208→  if (!providerId || !provider) {
   209→    return (
   210→      <div className="min-h-screen bg-background flex items-center justify-center p-4">
   211→        <div className="w-full max-w-sm">
   212→          <button
   213→            onClick={() => setView("home")}
   214→            className="text-muted-foreground hover:text-foreground text-sm mb-6 flex items-center gap-1"
   215→          >
   216→            <ArrowLeft className="w-4 h-4" /> Back
   217→          </button>
   218→
   219→          <div className="text-center mb-8">
   220→            <div className="w-16 h-16 rounded-2xl bg-emerald-500/15 border border-emerald-500/20 flex items-center justify-center mx-auto mb-3">
   221→              <Package className="w-8 h-8 text-emerald-400" />
   222→            </div>
   223→            <h1 className="text-xl font-semibold">Provider Login</h1>
   224→            <p className="text-xs text-muted-foreground mt-1">Enter your phone and 4-digit PIN</p>
   225→          </div>
   226→
   227→          <div className="space-y-3">
   228→            <div>
   229→              <label className="text-[11px] uppercase tracking-wider text-muted-foreground">Phone</label>
   230→              <input
   231→                value={loginPhone}
   232→                onChange={(e) => setLoginPhone(e.target.value)}
   233→                className="w-full mt-1 bg-card border border-border rounded-xl px-3 py-2.5 text-sm outline-none focus:border-emerald-500/40"
   234→                placeholder="+91..."
   235→              />
   236→            </div>
   237→            <div>
   238→              <label className="text-[11px] uppercase tracking-wider text-muted-foreground">4-digit PIN</label>
   239→              <input
   240→                value={loginPin}
   241→                onChange={(e) => setLoginPin(e.target.value)}
   242→                maxLength={4}
   243→                inputMode="numeric"
   244→                className="w-full mt-1 bg-card border border-border rounded-xl px-3 py-2.5 text-sm tracking-[0.5em] outline-none focus:border-emerald-500/40 font-mono"
   245→                placeholder="••••"
   246→              />
   247→            </div>
   248→            {loginError && (
   249→              <div className="text-xs text-rose-400 bg-rose-500/10 border border-rose-500/20 rounded-lg p-2">
   250→                {loginError}
   251→              </div>
   252→            )}
   253→            <button
   254→              onClick={handleLogin}
   255→              disabled={loginLoading || !loginPhone || !loginPin}
   256→              className="w-full bg-emerald-500 hover:bg-emerald-400 disabled:opacity-50 text-zinc-950 font-medium py-2.5 rounded-xl text-sm transition-colors"
   257→            >
   258→              {loginLoading ? "Signing in…" : "Sign in"}
   259→            </button>
   260→          </div>
   261→
   262→          <div className="mt-6 p-3 rounded-xl bg-card/60 border border-border/60 text-xs text-muted-foreground">
   263→            <p className="font-medium text-foreground mb-1">Demo providers</p>
   264→            <p>+919811100001 · +919811100002 · +919822200001</p>
   265→            <p className="mt-1">PIN: <span className="font-mono">1234</span></p>
   266→          </div>
   267→        </div>
   268→      </div>
   269→    );
   270→  }
   271→
   272→  // ── Render: onboarding ────────────────────────────────
   273→  if (view === "onboard") {
   274→    return <ProviderOnboarding provider={provider} onDone={() => setLocalView("home")} />;
   275→  }
   276→
   277→  // ── Render: incoming call screen (full-screen takeover) ─
   278→  if (incomingJob) {
   279→    return <IncomingCallScreen job={incomingJob} onAccept={handleAccept} onReject={handleReject} />;
   280→  }
   281→
   282→  // ── Render: job detail ────────────────────────────────
   283→  if (view === "job" && selectedJob) {
   284→    return (
   285→      <JobDetail
   286→        job={selectedJob}
   287→        onBack={() => { setLocalView("home"); setSelectedJob(null); }}
   288→        onUpdate={(s) => updateJobStatus(selectedJob, s)}
   289→      />
   290→    );
   291→  }
   292→
   293→  // ── Render: new manual job ────────────────────────────
   294→  if (view === "new") {
   295→    return (
   296→      <NewJobScreen
   297→        provider={provider}
   298→        onBack={() => setLocalView("home")}
   299→        onCreated={() => { setLocalView("home"); refresh(); }}
   300→      />
   301→    );
   302→  }
   303→
   304→  if (view === "history") {
   305→    return <HistoryScreen jobs={pastJobs} onBack={() => setLocalView("home")} />;
   306→  }
   307→
   308→  if (view === "custom") {
   309→    return <CustomRequestsScreen jobs={customRequests} provider={provider} onBack={() => setLocalView("home")} onSent={() => { setLocalView("home"); refresh(); }} />;
   310→  }
   311→
   312→  // ── Render: home ──────────────────────────────────────
   313→  return (
   314→    <div className="min-h-screen bg-background pb-20">
   315→      {/* Header */}
   316→      <header className="border-b border-border/60 bg-card/40 backdrop-blur-xl sticky top-0 z-10">
   317→        <div className="px-4 py-3 flex items-center justify-between">
   318→          <button onClick={() => setView("home")} className="text-muted-foreground hover:text-foreground">
   319→            <ArrowLeft className="w-5 h-5" />
   320→          </button>
   321→          <div className="text-center">
   322→            <p className="text-[11px] text-muted-foreground leading-none">{provider.tenantName}</p>
   323→            <p className="text-xs font-medium mt-0.5">{provider.cityName} · {provider.zone}</p>
   324→          </div>
   325→          <button onClick={() => clearProvider()} className="text-muted-foreground hover:text-foreground text-xs">
   326→            Logout
   327→          </button>
   328→        </div>
   329→      </header>
   330→
   331→      <div className="px-4 py-5 space-y-5 max-w-md mx-auto">
   332→        {/* Online toggle */}
   333→        <div className={cn(
   334→          "rounded-2xl p-5 border transition-colors",
   335→          provider.isOnline ? "bg-emerald-500/10 border-emerald-500/30" : "bg-card border-border"
   336→        )}>
   337→          <div className="flex items-center justify-between mb-3">
   338→            <div>
   339→              <p className="text-[11px] uppercase tracking-wider text-muted-foreground">Status</p>
   340→              <p className={cn("text-lg font-semibold mt-0.5", provider.isOnline ? "text-emerald-300" : "text-zinc-400")}>
   341→                {provider.isOnline ? "Online" : "Offline"}
   342→              </p>
   343→            </div>
   344→            <button
   345→              onClick={toggleOnline}
   346→              className={cn(
   347→                "relative w-14 h-8 rounded-full transition-colors",
   348→                provider.isOnline ? "bg-emerald-500" : "bg-zinc-700"
   349→              )}
   350→              aria-label="Toggle online"
   351→            >
   352→              <span className={cn(
   353→                "absolute top-1 w-6 h-6 rounded-full bg-white transition-transform",
   354→                provider.isOnline ? "translate-x-7" : "translate-x-1"
   355→              )} />
   356→            </button>
   357→          </div>
   358→          {provider.isOnline && (
   359→            <p className="text-xs text-emerald-300/80 flex items-center gap-1.5">
   360→              <span className="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-live-dot" />
   361→              Listening for jobs in {provider.zone}
   362→            </p>
   363→          )}
   364→        </div>
   365→
   366→        {/* Stats */}
   367→        <div className="grid grid-cols-3 gap-2">
   368→          <StatBox label="Today" value={String(activeJobs.length)} />
   369→          <StatBox label="Total jobs" value={String(provider.jobsDone)} />
   370→          <StatBox label="Earnings" value={formatINR(provider.earnings)} />
   371→        </div>
   372→
   373→        {/* Rating */}
   374→        <div className="flex items-center justify-between p-3 rounded-xl bg-card border border-border">
   375→          <div className="flex items-center gap-2">
   376→            <Star className="w-4 h-4 fill-amber-400 text-amber-400" />
   377→            <span className="text-sm font-medium">{provider.rating.toFixed(1)}</span>
   378→            <span className="text-xs text-muted-foreground">rating</span>
   379→          </div>
   380→          <span className="text-xs text-muted-foreground">Avg accept: {formatDuration(18)}</span>
   381→        </div>
   382→
   383→        {/* Quick actions */}
   384→        <div className="grid grid-cols-3 gap-2">
   385→          <QuickAction icon={Plus} label="New job" onClick={() => setLocalView("new")} />
   386→          <QuickAction icon={Bell} label="Requests" badge={customRequests.length} onClick={() => setLocalView("custom")} />
   387→          <QuickAction icon={History} label="History" onClick={() => setLocalView("history")} />
   388→        </div>
   389→
   390→        {/* Active jobs */}
   391→        <div>
   392→          <h3 className="text-[11px] uppercase tracking-wider text-muted-foreground mb-2">Active jobs</h3>
   393→          {activeJobs.length === 0 ? (
   394→            <EmptyState icon={Package} title="No active jobs" desc="New jobs will appear here" />
   395→          ) : (
   396→            <div className="space-y-2">
   397→              {activeJobs.map((job) => (
   398→                <JobCard key={job.id} job={job} onClick={() => { setSelectedJob(job); setLocalView("job"); }} />
   399→              ))}
   400→            </div>
   401→          )}
   402→        </div>
   403→      </div>
   404→    </div>
   405→  );
   406→}

   412→function StatBox({ label, value }: { label: string; value: string }) {
   413→  return (

   417→    </div>

   422→  return (
   423→    <button

   430→        <span className="absolute -top-1 -right-1 w-5 h-5 rounded-full bg-amber-500 text-zinc-950 text-[10px] font-semibold flex items-center justify-center">
   431→          {badge}
   432→        </span>
   433→      )}
   434→    </button>
   435→  );

   448→function JobCard({ job, onClick }: { job: Job; onClick: () => void }) {
   449→  const items = safeParse<OrderItem[]>(job.items, []);
   450→  const labels = job.service ? safeParse<Record<string, string>>(job.service.labels, {}) : {};
   451→  const svcName = job.service ? `${job.service.icon} ${labels.en || job.service.key}` : "Order";
   452→  return (
   453→    <button
   454→      onClick={onClick}
   455→      className="w-full text-left p-3 rounded-xl bg-card border border-border hover:border-emerald-500/30 transition-colors"
   456→    >
   457→      <div className="flex items-center justify-between mb-1">
   458→        <span className="text-xs font-medium">#{job.code}</span>
   459→        <span className={cn("text-[10px] px-1.5 py-0.5 rounded-full", ORDER_STATUS[job.status as keyof typeof ORDER_STATUS]?.bg, ORDER_STATUS[job.status as keyof typeof ORDER_STATUS]?.color)}>
   460→          {ORDER_STATUS[job.status as keyof typeof ORDER_STATUS]?.label || job.status}
   461→        </span>
   462→      </div>
   463→      <p className="text-sm font-medium mb-1">{svcName}</p>
   464→      {items.length > 0 && <p className="text-xs text-muted-foreground">{items.length} item(s) · {items[0]?.name}…</p>}
   465→      {job.description && <p className="text-xs text-muted-foreground truncate">{job.description}</p>}
   466→      <div className="flex items-center justify-between mt-2">
   467→        <span className="text-[11px] text-muted-foreground flex items-center gap-1">
   468→          <MapPin className="w-3 h-3" /> {job.addressArea || "—"}
   469→        </span>
   470→        <ChevronRight className="w-4 h-4 text-muted-foreground" />
   471→      </div>
   472→    </button>
   473→  );
   474→}

   476→function IncomingCallScreen({ job, onAccept, onReject }: { job: Job; onAccept: () => void; onReject: () => void }) {
   477→  const [seconds, setSeconds] = useState(45);
   478→  const items = safeParse<OrderItem[]>(job.items, []);
   479→  const labels = job.service ? safeParse<Record<string, string>>(job.service.labels, {}) : {};
   480→  const svcName = job.service ? `${job.service.icon} ${labels.en || job.service.key}` : "New Job";
   481→
   482→  useEffect(() => {
   483→    const id = setInterval(() => {
   484→      setSeconds((s) => {
   485→        if (s <= 1) {
   486→          clearInterval(id);
   487→          onReject();
   488→          return 0;
   489→        }
   490→        return s - 1;
   491→      });
   492→    }, 1000);
   493→    return () => clearInterval(id);
   494→  }, [onReject]);
   495→
   496→  // Vibrate if supported
   497→  useEffect(() => {
   498→    if (typeof navigator !== "undefined" && "vibrate" in navigator) {
   499→      navigator.vibrate([400, 200, 400, 200, 400]);
   500→      const id = setInterval(() => navigator.vibrate([400, 200, 400, 200, 400]), 1500);
   501→      return () => clearInterval(id);
   502→    }
   503→  }, []);
