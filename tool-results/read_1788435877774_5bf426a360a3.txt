   110→   110→function buildMenuList(tenantId: string, services: Array<{ id: string; key: string; kind: string; icon: string; labels: string }>, lang: "en" | "hi") {

   115→   115→      return order.indexOf(a.key) - order.indexOf(b.key);
   116→   116→    })
   117→   117→    .map((s) => {
   118→   118→      const labels = safeParse<Record<string, string>>(s.labels, {});
   119→   119→      return {
   120→   120→        id: `svc_${s.id}`,
   121→   121→        title: `${s.icon} ${labels[lang] || labels.en || s.key}`,
   122→   122→        description: s.kind === "order" ? "Order now" : s.kind === "book" ? "Book a slot" : s.kind === "team" ? "Talk to us" : "Tell us what you need",

  1005→  1005→function buildOrderSummary(
  1006→  1006→  svc: { icon: string; key: string; labels: string } | undefined,
  1007→  1007→  items: Array<{ name: string; qty?: string | number }>,
  1008→  1008→  shop: string | null,
  1009→  1009→  timing: string | null,
  1010→  1010→  address: string,
  1011→  1011→  area: string | null,
  1012→  1012→  lang: "en" | "hi"
  1013→  1013→): string {
  1014→  1014→  const labels = svc ? safeParse<Record<string, string>>(svc.labels, {}) : {};
  1015→  1015→  const svcName = svc ? `${svc.icon} ${labels[lang] || labels.en || svc.key}` : "Order";
  1016→  1016→  const lines = [
  1017→  1017→    `*${svcName}*`,
  1018→  1018→    "",
  1019→  1019→    "🛍️ Items:",
  1020→  1020→    ...items.map((it, i) => `  ${i + 1}. ${it.name}${it.qty ? ` ×${it.qty}` : ""}`),
  1021→  1021→  ];
  1022→  1022→  if (shop) lines.push("", `🏪 Shop: ${shop}`);
  1023→  1023→  if (timing) lines.push(`⏰ Timing: ${timing}`);
  1024→  1024→  lines.push("", `📍 Address:`, `  ${address}`);
  1025→  1025→  if (area) lines.push(`  Area: ${area}`);
  1026→  1026→  return lines.join("\n");
  1027→  1027→}
