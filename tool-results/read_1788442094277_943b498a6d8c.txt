   110→/** Merge grocery items by normalized name — sums quantities for duplicates (H9 fix) */
   111→function mergeItems(existing: Array<{ name: string; qty?: string | number }>, newItems: Array<{ name: string; qty?: string | number }>): Array<{ name: string; qty?: string | number }> {
   112→  const map = new Map<string, { name: string; qty?: string | number }>();
   113→  for (const it of [...existing, ...newItems]) {
   114→    const key = it.name.toLowerCase().trim();
   115→    const existingEntry = map.get(key);
   116→    if (existingEntry) {
   117→      // Try to sum quantities if both are numeric
   118→      const existingQty = parseFloat(String(existingEntry.qty || "0"));
   119→      const newQty = parseFloat(String(it.qty || "0"));
   120→      if (!isNaN(existingQty) && !isNaN(newQty)) {
   121→        existingEntry.qty = existingQty + newQty;
   122→      } else {
   123→        // Can't sum — keep the newer one
   124→        existingEntry.qty = it.qty || existingEntry.qty;
   125→      }
   126→    } else {
   127→      map.set(key, { ...it });
   128→    }
   129→  }
   130→  return Array.from(map.values());
   131→}

   133→function buildMenuList(tenantId: string, services: Array<{ id: string; key: string; kind: string; icon: string; labels: string }>, lang: "en" | "hi") {
   134→  const rows = services
   135→    .filter((s) => s.kind !== "custom" || s.key === "custom")
   136→    .sort((a, b) => {
   137→      const order = ["cake", "grocery", "chicken", "parcel", "ride", "repair", "team", "custom"];
   138→      return order.indexOf(a.key) - order.indexOf(b.key);
   139→    })
   140→    .map((s) => {
   141→      const labels = safeParse<Record<string, string>>(s.labels, {});
   142→      return {
   143→        id: `svc_${s.id}`,
   144→        title: `${s.icon} ${labels[lang] || labels.en || s.key}`,
   145→        description: s.kind === "order" ? "Order now" : s.kind === "book" ? "Book a slot" : s.kind === "team" ? "Talk to us" : "Tell us what you need",
   146→      };
   147→    });
   148→  return rows;
   149→}

   155→export async function POST(req: NextRequest) {
   156→  const body = await req.json();
   157→  const { tenantSlug, phone, message, button, location, mediaType } = body as {
   158→    tenantSlug: string;
   159→    phone: string;

   161→    button?: string;
   162→    location?: { lat: number; lng: number };
   163→    mediaType?: "voice" | "image";
   164→  };
   165→
   166→  if (!tenantSlug || !phone) {
   167→    return NextResponse.json({ error: "tenantSlug and phone required" }, { status: 400 });
   168→  }
   169→
   170→  const tenant = await db.tenant.findUnique({
   171→    where: { slug: tenantSlug },
   172→    include: {
   173→      services: { where: { isActive: true } },
   174→      cities: { where: { isActive: true } },
   175→    },
   176→  });
   177→  if (!tenant) return NextResponse.json({ error: "tenant not found" }, { status: 404 });
   178→
   179→  // Find or create customer
   180→  let customer = await db.customer.findUnique({
   181→    where: { tenantId_phone: { tenantId: tenant.id, phone } },
   182→  });
   183→  if (!customer) {
   184→    customer = await db.customer.create({
   185→      data: { tenantId: tenant.id, phone, language: "en" },
   186→    });
   187→  }
   188→
   189→  // Find or create session
   190→  let session = await db.botSession.findUnique({
   191→    where: { tenantId_phone: { tenantId: tenant.id, phone } },
   192→  });
   193→  if (!session) {
   194→    session = await db.botSession.create({
   195→      data: { tenantId: tenant.id, customerId: customer.id, phone },
   196→    });
   197→  }
   198→
   199→  const lang = (customer.language as "en" | "hi") || "en";
   200→  const replies: BotReply[] = [];
   201→
   202→  // ── Handle charges agree/cancel buttons (sent by provider via /charges) ──
   203→  if (button && button.startsWith("charges_agree_")) {
   204→    const orderId = button.slice("charges_agree_".length);
   205→    try {
   206→      const { agreeToCharges } = await import("@/lib/order-actions");
   207→      await agreeToCharges(orderId, tenant.id);
   208→      replies.push({ kind: "text", text: "✅ Charges agreed. The provider will send a payment link shortly." });
   209→    } catch {
   210→      replies.push({ kind: "text", text: "Something went wrong. Please try again." });
   211→    }
   212→    return NextResponse.json({ replies, session: { state: session.state } });
   213→  }
   214→  if (button && button.startsWith("charges_cancel_")) {
   215→    const orderId = button.slice("charges_cancel_".length);
   216→    try {
   217→      const { declineCharges } = await import("@/lib/order-actions");
   218→      await declineCharges(orderId, tenant.id);
   219→      replies.push({ kind: "text", text: "❌ Order cancelled. Type *menu* to start a new order." });
   220→    } catch {
   221→      replies.push({ kind: "text", text: "Something went wrong." });
   222→    }
   223→    return NextResponse.json({ replies, session: { state: session.state } });
   224→  }
   225→
   226→  // ── Handle rating buttons (rate_5_<orderId>, rate_4_, rate_3_) ──
   227→  if (button && button.startsWith("rate_")) {
   228→    const match = button.match(/^rate_(\d+)_(.+)$/);
   229→    if (match) {
   230→      const rating = parseInt(match[1], 10);
   231→      const orderId = match[2];
   232→      try {
   233→        const { submitReview } = await import("@/lib/order-actions");
   234→        const result = await submitReview(orderId, tenant.id, rating);
   235→        if (result.ok) {
   236→          replies.push({ kind: "text", text: `🙏 Thank you for your ${rating}★ rating!${result.googleReviewUrl ? "\n\nWe've also sent you a link to review us on Google — it really helps!" : ""}` });

   238→          replies.push({ kind: "text", text: "You've already rated this order. Thank you! 🙏" });
   239→        } else {
   240→          replies.push({ kind: "text", text: "Thanks for your feedback! 🙏" });
   241→        }
   242→      } catch {
   243→        replies.push({ kind: "text", text: "Thanks for your feedback! 🙏" });
   244→      }
   245→      return NextResponse.json({ replies, session: { state: session.state } });
   246→    }
   247→  }
   248→
   249→  // ── Handle quote accept/decline buttons ──
   250→  if (button && button.startsWith("quote_accept_")) {
   251→    const orderId = button.slice("quote_accept_".length);
   252→    try {
   253→      const { acceptQuote } = await import("@/lib/order-actions");
   254→      const result = await acceptQuote(orderId, tenant.id);
   255→      if (result.ok) {
   256→        replies.push({ kind: "text", text: "✅ Quote accepted! Your order is confirmed. The provider will reach out shortly." });

   298→      kind: "list",

   306→

   310→      where: { id: session.id },
   311→      data: {
   312→        state: "menu",
   313→        draftService: null,
   314→        draftItems: "[]",
   315→        draftTiming: null,
   316→        draftShop: null,
   317→        draftAddress: null,
   318→        draftLat: null,
   319→        draftLng: null,

   321→    });

   324→      kind: "list",
   325→      text: t(tenant, lang, "menu_title"),

   881→        const updated = [...addresses.filter((a) => a.area !== area), newAddr];
   882→        await db.customer.update({
   883→          where: { id: customer.id },
   884→          data: { addresses: JSON.stringify(updated) },
   885→        });
   886→        await db.botSession.update({ where: { id: session.id }, data: { state: "confirm" } });
   887→        const items = safeParse<{ name: string; qty?: string | number }[]>(session.draftItems, []);
   888→        const svc = tenant.services.find((s) => s.id === session.draftService);
   889→        const summary = buildOrderSummary(svc, items, session.draftShop, session.draftTiming, session.draftAddress || "", area, lang);
   890→        replies.push({
   891→          kind: "buttons",
   892→          text: `${t(tenant, lang, "confirm_title")}\n\n${summary}\n\n${t(tenant, lang, "confirm_ask")}`,
   893→          buttons: [
   894→            { id: "confirm_yes", label: "✅ Confirm" },
   895→            { id: "confirm_change", label: "✏️ Change" },
   896→            { id: "cancel", label: "❌ Cancel" },
   897→          ],
   898→        });
   899→        break;
   900→      }
   901→      if (button === "addr_type") {
   902→        replies.push({ kind: "text", text: "Type your address with landmark:" });
   903→        await db.botSession.update({ where: { id: session.id }, data: { state: "address_new" } });
   904→        break;
   905→      }
   906→      break;
   907→    }
   908→
   909→    case "confirm": {
   910→      if (button === "confirm_yes") {
   911→        // ── PLACE ORDER ──────────────────────────────────
   912→        const items = safeParse<{ name: string; qty?: string | number }[]>(session.draftItems, []);
   913→        const svc = tenant.services.find((s) => s.id === session.draftService);
   914→        const lastOrder = await db.order.findFirst({
   915→          where: { tenantId: tenant.id },
   916→          orderBy: { code: "desc" },
   917→        });
   918→        const code = lastOrder ? String(parseInt(lastOrder.code, 10) + 1) : "1001";
   919→        const cityId = tenant.cities[0]?.id;
   920→        if (!cityId) {
   921→          replies.push({ kind: "text", text: "Sorry, no active city. Please contact support." });
   922→          break;
   923→        }
   924→        const order = await db.order.create({
   925→          data: {
   926→            tenantId: tenant.id,
   927→            cityId,
   928→            customerId: customer.id,
   929→            serviceId: svc?.id || null,
   930→            code,
   931→            status: "new",
   932→            kind: svc?.kind || "order",
   933→            items: JSON.stringify(items),
   934→            description: items[0]?.name || null,
   935→            preferredShop: session.draftShop,
   936→            timing: session.draftTiming,
   937→            addressText: session.draftAddress,
   938→            addressArea: session.draftAddress ? session.draftAddress.split(" (")[0] : null,
   939→            addressLat: session.draftLat,
   940→            addressLng: session.draftLng,
   941→            source: "bot",
   942→            activity: {
   943→              create: [
   944→                { tenantId: tenant.id, actor: "bot", action: "created", detail: "Order created via WhatsApp" },
   945→              ],
   946→            },
   947→          },
   948→        });
   949→        // Broadcast to matching providers
   950→        const matchingProviders = await db.provider.findMany({
   951→          where: {
   952→            tenantId: tenant.id,
   953→            cityId,
   954→            isOnline: true,
   955→            isActive: true,
   956→          },
   957→        });
   958→        const toBroadcast = svc
   959→          ? matchingProviders.filter((p) => {
   960→              const sids = safeParse<string[]>(p.serviceIds, []);
   961→              return sids.length === 0 || sids.includes(svc.id);
   962→            })
   963→          : matchingProviders;
   964→
   965→        if (toBroadcast.length > 0) {
   966→          await db.order.update({ where: { id: order.id }, data: { status: "broadcast" } });
   967→          await db.orderBroadcast.createMany({
   968→            data: toBroadcast.map((p) => ({ orderId: order.id, providerId: p.id, status: "pending" })),
   969→          });
   970→          await db.activity.create({
   971→            data: {
   972→              tenantId: tenant.id, orderId: order.id,
   973→              actor: "system", action: "broadcast",
   974→              detail: `Broadcast to ${toBroadcast.length} provider(s)`,
   975→            },
   976→          });
   977→        }
   978→
   979→        await db.botSession.update({
   980→          where: { id: session.id },
   981→          data: {
   982→            state: "done",
   983→            draftService: null,
   984→            draftItems: "[]",
   985→            draftShop: null,
   986→            draftTiming: null,
   987→            draftAddress: null,
   988→            draftLat: null,
   989→            draftLng: null,
   990→          },
   991→        });
   992→        replies.push({ kind: "text", text: t(tenant, lang, "placed", { code }) });
   993→        // Show menu again
   994→        const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   995→        replies.push({
   996→          kind: "list",
   997→          text: t(tenant, lang, "menu_title"),
   998→          listTitle: t(tenant, lang, "menu_title"),
   999→          listButton: t(tenant, lang, "menu_button"),
  1000→          sections: [{ title: "Services", rows: menuRows }],
  1001→        });
  1002→        return NextResponse.json({ replies, session: { state: "done" }, order });
  1003→      }
  1004→      if (button === "confirm_change") {
  1005→        // Restart from service
  1006→        await db.botSession.update({
  1007→          where: { id: session.id },
  1008→          data: { state: "service_draft", draftItems: "[]" },
  1009→        });
  1010→        replies.push({ kind: "text", text: "Let's start over. Tell us what you need:" });
  1011→        break;
  1012→      }
  1013→      if (button === "cancel") {
  1014→        await db.botSession.update({
  1015→          where: { id: session.id },
  1016→          data: { state: "menu", draftService: null, draftItems: "[]" },
  1017→        });
  1018→        replies.push({ kind: "text", text: t(tenant, lang, "cancelled") });
  1019→        const menuRows = buildMenuList(tenant.id, tenant.services, lang);
  1020→        replies.push({
  1021→          kind: "list",
  1022→          text: t(tenant, lang, "menu_title"),
  1023→          listTitle: t(tenant, lang, "menu_title"),
  1024→          listButton: t(tenant, lang, "menu_button"),
  1025→          sections: [{ title: "Services", rows: menuRows }],
  1026→        });
  1027→      }
  1028→      break;
  1029→    }
  1030→
  1031→    case "done":
  1032→    default: {
  1033→      // Show menu
  1034→      const menuRows = buildMenuList(tenant.id, tenant.services, lang);
  1035→      await db.botSession.update({ where: { id: session.id }, data: { state: "menu" } });
  1036→      replies.push({
  1037→        kind: "list",
  1038→        text: t(tenant, lang, "menu_title"),
  1039→        listTitle: t(tenant, lang, "menu_title"),
  1040→        listButton: t(tenant, lang, "menu_button"),
  1041→        sections: [{ title: "Services", rows: menuRows }],
  1042→      });
  1043→      break;
  1044→    }
  1045→  }
  1046→
  1047→  // Update session lastMessageAt
  1048→  await db.botSession.update({
  1049→    where: { id: session.id },
  1050→    data: { lastMessageAt: new Date() },
  1051→  });
  1052→
  1053→  return NextResponse.json({ replies, session: { state: session.state } });
  1054→}

  1056→function buildOrderSummary(
  1057→  svc: { icon: string; key: string; labels: string } | undefined,
  1058→  items: Array<{ name: string; qty?: string | number }>,
  1059→  shop: string | null,
  1060→  timing: string | null,
  1061→  address: string,
  1062→  area: string | null,
  1063→  lang: "en" | "hi"
  1064→): string {
  1065→  const labels = svc ? safeParse<Record<string, string>>(svc.labels, {}) : {};
  1066→  const svcName = svc ? `${svc.icon} ${labels[lang] || labels.en || svc.key}` : "Order";
  1067→  const lines = [
  1068→    `*${svcName}*`,
  1069→    "",
  1070→    "🛍️ Items:",
  1071→    ...items.map((it, i) => `  ${i + 1}. ${it.name}${it.qty ? ` ×${it.qty}` : ""}`),
  1072→  ];
  1073→  if (shop) lines.push("", `🏪 Shop: ${shop}`);
  1074→  if (timing) lines.push(`⏰ Timing: ${timing}`);
  1075→  lines.push("", `📍 Address:`, `  ${address}`);
  1076→  if (area) lines.push(`  Area: ${area}`);
  1077→  return lines.join("\n");
  1078→}
