   766→   766→          replies.push({ kind: "text", text: "Sorry, no active city. Please contact support." });
   767→   767→          break;
   768→   768→        }
   769→   769→        const order = await db.order.create({
   770→   770→          data: {
   771→   771→            tenantId: tenant.id,
   772→   772→            cityId,
   773→   773→            customerId: customer.id,
   774→   774→            serviceId: svc?.id || null,
   775→   775→            code,
   776→   776→            status: "new",
   777→   777→            kind: svc?.kind || "order",
   778→   778→            items: JSON.stringify(items),
   779→   779→            description: items[0]?.name || null,
   780→   780→            preferredShop: session.draftShop,
   781→   781→            timing: session.draftTiming,
   782→   782→            addressText: session.draftAddress,
   783→   783→            addressArea: session.draftAddress ? session.draftAddress.split(" (")[0] : null,
   784→   784→            addressLat: session.draftLat,
   785→   785→            addressLng: session.draftLng,
   786→   786→            source: "bot",
   787→   787→            activity: {
   788→   788→              create: [
   789→   789→                { tenantId: tenant.id, actor: "bot", action: "created", detail: "Order created via WhatsApp" },
   790→   790→              ],
   791→   791→            },
   792→   792→          },
   793→   793→        });
   794→   794→        // Broadcast to matching providers
   795→   795→        const matchingProviders = await db.provider.findMany({
   796→   796→          where: {
   797→   797→            tenantId: tenant.id,
   798→   798→            cityId,
   799→   799→            isOnline: true,
   800→   800→            isActive: true,
   801→   801→          },
   802→   802→        });
   803→   803→        const toBroadcast = svc
   804→   804→          ? matchingProviders.filter((p) => {
   805→   805→              const sids = safeParse<string[]>(p.serviceIds, []);
   806→   806→              return sids.length === 0 || sids.includes(svc.id);
   807→   807→            })
   808→   808→          : matchingProviders;
   809→   809→
   810→   810→        if (toBroadcast.length > 0) {
   811→   811→          await db.order.update({ where: { id: order.id }, data: { status: "broadcast" } });
   812→   812→          await db.orderBroadcast.createMany({
   813→   813→            data: toBroadcast.map((p) => ({ orderId: order.id, providerId: p.id, status: "pending" })),
   814→   814→          });
   815→   815→          await db.activity.create({
   816→   816→            data: {
   817→   817→              tenantId: tenant.id, orderId: order.id,
   818→   818→              actor: "system", action: "broadcast",
   819→   819→              detail: `Broadcast to ${toBroadcast.length} provider(s)`,
   820→   820→            },
   821→   821→          });
   822→   822→        }
   823→   823→
   824→   824→        await db.botSession.update({
   825→   825→          where: { id: session.id },
   826→   826→          data: {
   827→   827→            state: "done",
   828→   828→            draftService: null,
   829→   829→            draftItems: "[]",
   830→   830→            draftShop: null,
   831→   831→            draftTiming: null,
   832→   832→            draftAddress: null,
   833→   833→            draftLat: null,
   834→   834→            draftLng: null,
   835→   835→          },
   836→   836→        });
   837→   837→        replies.push({ kind: "text", text: t(tenant, lang, "placed", { code }) });
   838→   838→        // Show menu again
   839→   839→        const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   840→   840→        replies.push({
   841→   841→          kind: "list",
   842→   842→          text: t(tenant, lang, "menu_title"),
   843→   843→          listTitle: t(tenant, lang, "menu_title"),
   844→   844→          listButton: t(tenant, lang, "menu_button"),
   845→   845→          sections: [{ title: "Services", rows: menuRows }],
   846→   846→        });
   847→   847→        return NextResponse.json({ replies, session: { state: "done" }, order });
   848→   848→      }
   849→   849→      if (button === "confirm_change") {
   850→   850→        // Restart from service
   851→   851→        await db.botSession.update({
   852→   852→          where: { id: session.id },
   853→   853→          data: { state: "service_draft", draftItems: "[]" },
   854→   854→        });
   855→   855→        replies.push({ kind: "text", text: "Let's start over. Tell us what you need:" });
   856→   856→        break;
   857→   857→      }
   858→   858→      if (button === "cancel") {
   859→   859→        await db.botSession.update({
   860→   860→          where: { id: session.id },
   861→   861→          data: { state: "menu", draftService: null, draftItems: "[]" },
   862→   862→        });
   863→   863→        replies.push({ kind: "text", text: t(tenant, lang, "cancelled") });
   864→   864→        const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   865→   865→        replies.push({
   866→   866→          kind: "list",
   867→   867→          text: t(tenant, lang, "menu_title"),
   868→   868→          listTitle: t(tenant, lang, "menu_title"),
   869→   869→          listButton: t(tenant, lang, "menu_button"),
   870→   870→          sections: [{ title: "Services", rows: menuRows }],
   871→   871→        });
   872→   872→      }
   873→   873→      break;
   874→   874→    }
   875→   875→
   876→   876→    case "done":
   877→   877→    default: {
   878→   878→      // Show menu
   879→   879→      const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   880→   880→      await db.botSession.update({ where: { id: session.id }, data: { state: "menu" } });
   881→   881→      replies.push({
   882→   882→        kind: "list",
   883→   883→        text: t(tenant, lang, "menu_title"),
   884→   884→        listTitle: t(tenant, lang, "menu_title"),
   885→   885→        listButton: t(tenant, lang, "menu_button"),
   886→   886→        sections: [{ title: "Services", rows: menuRows }],
   887→   887→      });
   888→   888→      break;
   889→   889→    }
   890→   890→  }
   891→   891→
   892→   892→  // Update session lastMessageAt
   893→   893→  await db.botSession.update({
   894→   894→    where: { id: session.id },
   895→   895→    data: { lastMessageAt: new Date() },
   896→   896→  });
   897→   897→
   898→   898→  return NextResponse.json({ replies, session: { state: session.state } });
   899→   899→}

   901→   901→function buildOrderSummary(
   902→   902→  svc: { icon: string; key: string; labels: string } | undefined,
   903→   903→  items: Array<{ name: string; qty?: string | number }>,
   904→   904→  shop: string | null,
   905→   905→  timing: string | null,
   906→   906→  address: string,
   907→   907→  area: string | null,
   908→   908→  lang: "en" | "hi"
   909→   909→): string {
   910→   910→  const labels = svc ? safeParse<Record<string, string>>(svc.labels, {}) : {};
   911→   911→  const svcName = svc ? `${svc.icon} ${labels[lang] || labels.en || svc.key}` : "Order";
   912→   912→  const lines = [
   913→   913→    `*${svcName}*`,
   914→   914→    "",
   915→   915→    "🛍️ Items:",
   916→   916→    ...items.map((it, i) => `  ${i + 1}. ${it.name}${it.qty ? ` ×${it.qty}` : ""}`),
   917→   917→  ];
   918→   918→  if (shop) lines.push("", `🏪 Shop: ${shop}`);
   919→   919→  if (timing) lines.push(`⏰ Timing: ${timing}`);
   920→   920→  lines.push("", `📍 Address:`, `  ${address}`);
   921→   921→  if (area) lines.push(`  Area: ${area}`);
   922→   922→  return lines.join("\n");
   923→   923→}
