     1→/**
     2→ * POST /api/bot/send
     3→ * Body: { tenantSlug, phone, message?: text, button?: id, location?: {lat,lng}, mediaType?: "voice"|"image" }

     5→ * Returns: { replies: BotReply[], session: {...}, order?: {...} }
     6→ *
     7→ * BotReply = { kind: "text" | "buttons" | "list", text, buttons?, sections? }
     8→ *
     9→ * State machine: language → menu → service_draft → address → confirm → done
    10→ * Resilient: "cancel"/"menu" anywhere → back to menu. Old button taps → re-show current step.
    11→ */

   110→function buildMenuList(tenantId: string, services: Array<{ id: string; key: string; kind: string; icon: string; labels: string }>, lang: "en" | "hi") {
   111→  const rows = services
   112→    .filter((s) => s.kind !== "custom" || s.key === "custom")
   113→    .sort((a, b) => {
   114→      const order = ["cake", "grocery", "chicken", "parcel", "ride", "repair", "team", "custom"];
   115→      return order.indexOf(a.key) - order.indexOf(b.key);
   116→    })
   117→    .map((s) => {
   118→      const labels = safeParse<Record<string, string>>(s.labels, {});
   119→      return {
   120→        id: `svc_${s.id}`,
   121→        title: `${s.icon} ${labels[lang] || labels.en || s.key}`,
   122→        description: s.kind === "order" ? "Order now" : s.kind === "book" ? "Book a slot" : s.kind === "team" ? "Talk to us" : "Tell us what you need",
   123→      };
   124→    });
   125→  return rows;
   126→}

   132→export async function POST(req: NextRequest) {
   133→  const body = await req.json();
   134→  const { tenantSlug, phone, message, button, location, mediaType } = body as {
   135→    tenantSlug: string;
   136→    phone: string;
   137→    message?: string;
   138→    button?: string;
   139→    location?: { lat: number; lng: number };
   140→    mediaType?: "voice" | "image";
   141→  };
   142→
   143→  if (!tenantSlug || !phone) {
   144→    return NextResponse.json({ error: "tenantSlug and phone required" }, { status: 400 });
   145→  }
   146→
   147→  const tenant = await db.tenant.findUnique({
   148→    where: { slug: tenantSlug },
   149→    include: {
   150→      services: { where: { isActive: true } },
   151→      cities: { where: { isActive: true } },
   152→    },
   153→  });
   154→  if (!tenant) return NextResponse.json({ error: "tenant not found" }, { status: 404 });
   155→
   156→  // Find or create customer
   157→  let customer = await db.customer.findUnique({
   158→    where: { tenantId_phone: { tenantId: tenant.id, phone } },
   159→  });
   160→  if (!customer) {
   161→    customer = await db.customer.create({
   162→      data: { tenantId: tenant.id, phone, language: "en" },
   163→    });
   164→  }
   165→
   166→  // Find or create session
   167→  let session = await db.botSession.findUnique({
   168→    where: { tenantId_phone: { tenantId: tenant.id, phone } },
   169→  });
   170→  if (!session) {
   171→    session = await db.botSession.create({
   172→      data: { tenantId: tenant.id, customerId: customer.id, phone },
   173→    });
   174→  }
   175→
   176→  const lang = (customer.language as "en" | "hi") || "en";
   177→  const replies: BotReply[] = [];
   178→
   179→  // ── Handle "cancel" / "menu" anywhere ────────────────────
   180→  const text = (message || "").trim().toLowerCase();
   181→  if (text === "cancel" || text === "❌ cancel" || text === "cancel order") {
   182→    // Reset draft
   183→    await db.botSession.update({
   184→      where: { id: session.id },
   185→      data: {
   186→        state: "menu",
   187→        draftService: null,
   188→        draftItems: "[]",
   189→        draftTiming: null,
   190→        draftShop: null,
   191→        draftAddress: null,
   192→        draftLat: null,
   193→        draftLng: null,
   194→      },
   195→    });
   196→    replies.push({ kind: "text", text: t(tenant, lang, "cancelled") });
   197→    // Show menu
   198→    const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   199→    replies.push({
   200→      kind: "list",
   201→      text: t(tenant, lang, "menu_title"),
   202→      listTitle: t(tenant, lang, "menu_title"),
   203→      listButton: t(tenant, lang, "menu_button"),
   204→      sections: [{ title: "Services", rows: menuRows }],
   205→    });
   206→    return NextResponse.json({ replies, session: { state: "menu" } });
   207→  }
   208→
   209→  if (text === "menu" || text === "start" || text === "hi" || text === "hello") {
   210→    await db.botSession.update({
   211→      where: { id: session.id },
   212→      data: { state: "menu", draftService: null, draftItems: "[]" },
   213→    });
   214→    const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   215→    replies.push({
   216→      kind: "list",
   217→      text: t(tenant, lang, "menu_title"),
   218→      listTitle: t(tenant, lang, "menu_title"),
   219→      listButton: t(tenant, lang, "menu_button"),
   220→      sections: [{ title: "Services", rows: menuRows }],
   221→    });
   222→    return NextResponse.json({ replies, session: { state: "menu" } });
   223→  }
   224→
   225→  // ── State machine ────────────────────────────────────────
   226→  switch (session.state) {
   227→    case "language": {
   228→      if (button === "lang_en") {
   229→        await db.customer.update({ where: { id: customer.id }, data: { language: "en" } });
   230→        const menuRows = buildMenuList(tenant.id, tenant.services, "en");
   231→        await db.botSession.update({ where: { id: session.id }, data: { state: "menu" } });
   232→        replies.push({
   233→          kind: "list",
   234→          text: t(tenant, "en", "menu_title"),
   235→          listTitle: t(tenant, "en", "menu_title"),
   236→          listButton: t(tenant, "en", "menu_button"),
   237→          sections: [{ title: "Services", rows: menuRows }],
   238→        });
   239→      } else if (button === "lang_hi") {
   240→        await db.customer.update({ where: { id: customer.id }, data: { language: "hi" } });
   241→        const menuRows = buildMenuList(tenant.id, tenant.services, "hi");
   242→        await db.botSession.update({ where: { id: session.id }, data: { state: "menu" } });
   243→        replies.push({
   244→          kind: "list",
   245→          text: t(tenant, "hi", "menu_title"),
   246→          listTitle: t(tenant, "hi", "menu_title"),
   247→          listButton: t(tenant, "hi", "menu_button"),
   248→          sections: [{ title: "Services", rows: menuRows }],
   249→        });
   250→      } else {
   251→        replies.push({
   252→          kind: "buttons",
   253→          text: t(tenant, lang, "welcome"),
   254→          buttons: [
   255→            { id: "lang_en", label: "🇬🇧 English" },
   256→            { id: "lang_hi", label: "🇮🇳 हिन्दी" },
   257→          ],
   258→        });
   259→      }
   260→      break;
   261→    }
   262→
   263→    case "menu": {
   264→      // Handle list item tap: svc_<id>
   265→      if (button && button.startsWith("svc_")) {
   266→        const svcId = button.slice(4);
   267→        const svc = tenant.services.find((s) => s.id === svcId);
   268→        if (!svc) {
   269→          replies.push({ kind: "text", text: t(tenant, lang, "invalid") });
   270→          break;
   271→        }
   272→        await db.botSession.update({
   273→          where: { id: session.id },
   274→          data: { state: "service_draft", draftService: svc.id, draftItems: "[]", draftShop: null, draftTiming: null },
   275→        });
   276→        // Service-specific first message
   277→        if (svc.key === "grocery") {
   278→          replies.push({ kind: "buttons", text: t(tenant, lang, "ask_grocery"), buttons: [
   279→            { id: "grocery_done", label: "✅ Done" },
   280→            { id: "grocery_more", label: "➕ Add more" },
   281→            { id: "cancel", label: "❌ Start over" },
   282→          ]});
   283→        } else if (svc.kind === "order") {
   284→          replies.push({ kind: "text", text: t(tenant, lang, `ask_${svc.key}`) });
   285→        } else if (svc.kind === "book") {
   286→          replies.push({ kind: "text", text: t(tenant, lang, `ask_${svc.key}`) });
   287→        } else if (svc.kind === "team") {
   288→          replies.push({ kind: "text", text: t(tenant, lang, "ask_team") });
   289→          // Mark as custom
   290→          await db.botSession.update({
   291→            where: { id: session.id },
   292→            data: { state: "service_draft", draftService: svc.id },
   293→          });
   294→        } else if (svc.kind === "custom") {
   295→          replies.push({ kind: "text", text: t(tenant, lang, "ask_custom") });
   296→        }
   297→      } else {
   298→        // Re-show menu
   299→        const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   300→        replies.push({
   301→          kind: "list",
   302→          text: t(tenant, lang, "old_button") + "\n" + t(tenant, lang, "menu_title"),
   303→          listTitle: t(tenant, lang, "menu_title"),
   304→          listButton: t(tenant, lang, "menu_button"),
   305→          sections: [{ title: "Services", rows: menuRows }],
   306→        });
   307→      }
   308→      break;
   309→    }
   310→
   311→    case "service_draft": {
   312→      const svcId = session.draftService;
   313→      if (!svcId) {
   314→        // Reset
   315→        await db.botSession.update({ where: { id: session.id }, data: { state: "menu" } });
   316→        replies.push({ kind: "text", text: t(tenant, lang, "back_to_menu") });
   317→        break;
   318→      }
   319→      const svc = tenant.services.find((s) => s.id === svcId);
   320→      if (!svc) {
   321→        await db.botSession.update({ where: { id: session.id }, data: { state: "menu", draftService: null } });
   322→        replies.push({ kind: "text", text: t(tenant, lang, "back_to_menu") });
   323→        break;
   324→      }
   325→
   326→      // ── Grocery: accumulate items until Done ──────────
   327→      if (svc.key === "grocery") {
   328→        if (button === "grocery_done") {
   329→          const items = safeParse<{ name: string; qty?: string | number }[]>(session.draftItems, []);
   330→          if (items.length === 0) {
   331→            replies.push({ kind: "text", text: "Your list is empty. Add items first." });
   332→            break;
   333→          }
   334→          replies.push({ kind: "text", text: t(tenant, lang, "grocery_done") });
   335→          // Move to shop question
   336→          await db.botSession.update({
   337→            where: { id: session.id },
   338→            data: { state: "shop" },
   339→          });
   340→          replies.push({
   341→            kind: "buttons",
   342→            text: t(tenant, lang, "ask_shop"),
   343→            buttons: [
   344→              { id: "shop_any", label: "🏪 Any shop" },
   345→              { id: "shop_specify", label: "✏️ Specify" },
   346→            ],
   347→          });
   348→          break;
   349→        }
   350→        if (button === "grocery_more" || message || mediaType) {
   351→          const items = safeParse<{ name: string; qty?: string | number }[]>(session.draftItems, []);
   352→          // If media, simulate AI extraction
   353→          if (mediaType === "voice") {
   354→            replies.push({ kind: "text", text: t(tenant, lang, "voice_received") });
   355→            // Simulated extraction
   356→            const fakeItems = [
   357→              { name: "Rice 5kg", qty: 1 },
   358→              { name: "Cooking Oil 1L", qty: 2 },
   359→              { name: "Sugar 1kg", qty: 1 },
   360→            ];
   361→            const merged = [...items, ...fakeItems];
   362→            await db.botSession.update({
   363→              where: { id: session.id },
   364→              data: { draftItems: JSON.stringify(merged) },
   365→            });
   366→            const summary = merged.map((it, i) => `${i + 1}. ${it.name}${it.qty ? ` ×${it.qty}` : ""}`).join("\n");
   367→            replies.push({ kind: "buttons", text: `${t(tenant, lang, "ask_grocery_summary")}\n\n${summary}`, buttons: [
   368→              { id: "grocery_done", label: "✅ Done" },
   369→              { id: "grocery_more", label: "➕ Add more" },
   370→              { id: "cancel", label: "❌ Start over" },
   371→            ]});
   372→            break;
   373→          }
   374→          if (mediaType === "image") {
   375→            replies.push({ kind: "text", text: t(tenant, lang, "photo_received") });
   376→            const fakeItems = [
   377→              { name: "Atta 10kg", qty: 1 },
   378→              { name: "Tea 250g", qty: 1 },
   379→              { name: "Biscuits", qty: 3 },
   380→            ];
   381→            const merged = [...items, ...fakeItems];
   382→            await db.botSession.update({
   383→              where: { id: session.id },
   384→              data: { draftItems: JSON.stringify(merged) },
   385→            });
   386→            const summary = merged.map((it, i) => `${i + 1}. ${it.name}${it.qty ? ` ×${it.qty}` : ""}`).join("\n");
   387→            replies.push({ kind: "buttons", text: `${t(tenant, lang, "ask_grocery_summary")}\n\n${summary}`, buttons: [
   388→              { id: "grocery_done", label: "✅ Done" },
   389→              { id: "grocery_more", label: "➕ Add more" },
   390→              { id: "cancel", label: "❌ Start over" },
   391→            ]});
   392→            break;
   393→          }
   394→          // Parse typed items (split by newlines or commas)
   395→          if (message) {
   396→            const lines = message.split(/\n|,|;/).map((l) => l.trim()).filter(Boolean);
   397→            const newItems = lines.map((l) => {
   398→              const m = l.match(/^(\d+\.?\d*)\s*(?:x|×|kg|g|l|ltr|pcs|piece|pkt|pack)?\s*(.*)/i);
   399→              if (m && m[2]) return { name: m[2], qty: m[1] };
   400→              return { name: l, qty: 1 };
   401→            });
   402→            const merged = [...items, ...newItems];
   403→            await db.botSession.update({
   404→              where: { id: session.id },
   405→              data: { draftItems: JSON.stringify(merged) },

   856→        break;
   857→      }
   858→      if (button === "cancel") {
   859→        await db.botSession.update({
   860→          where: { id: session.id },
   861→          data: { state: "menu", draftService: null, draftItems: "[]" },

   864→        const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   865→        replies.push({
   866→          kind: "list",
   867→          text: t(tenant, lang, "menu_title"),

   869→          listButton: t(tenant, lang, "menu_button"),
   870→          sections: [{ title: "Services", rows: menuRows }],

   872→      }
   873→      break;
   874→    }
   875→
   876→    case "done":
   877→    default: {
   878→      // Show menu
   879→      const menuRows = buildMenuList(tenant.id, tenant.services, lang);
   880→      await db.botSession.update({ where: { id: session.id }, data: { state: "menu" } });
   881→      replies.push({
   882→        kind: "list",
   883→        text: t(tenant, lang, "menu_title"),
   884→        listTitle: t(tenant, lang, "menu_title"),
   885→        listButton: t(tenant, lang, "menu_button"),
   886→        sections: [{ title: "Services", rows: menuRows }],
   887→      });
   888→      break;
   889→    }
   890→  }
   891→
   892→  // Update session lastMessageAt
   893→  await db.botSession.update({
   894→    where: { id: session.id },
   895→    data: { lastMessageAt: new Date() },
   896→  });
   897→
   898→  return NextResponse.json({ replies, session: { state: session.state } });
   899→}
   900→
   901→function buildOrderSummary(
   902→  svc: { icon: string; key: string; labels: string } | undefined,
   903→  items: Array<{ name: string; qty?: string | number }>,
   904→  shop: string | null,
   905→  timing: string | null,
   906→  address: string,
   907→  area: string | null,
   908→  lang: "en" | "hi"
   909→): string {
   910→  const labels = svc ? safeParse<Record<string, string>>(svc.labels, {}) : {};
   911→  const svcName = svc ? `${svc.icon} ${labels[lang] || labels.en || svc.key}` : "Order";
   912→  const lines = [
   913→    `*${svcName}*`,
   914→    "",
   915→    "🛍️ Items:",
   916→    ...items.map((it, i) => `  ${i + 1}. ${it.name}${it.qty ? ` ×${it.qty}` : ""}`),
