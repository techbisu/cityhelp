    62→export function ProviderApp() {
    63→  const setView = useApp((s) => s.setView);
    64→  const providerId = useApp((s) => s.providerId);
    65→  const providerTenantSlug = useApp((s) => s.providerTenantSlug);
    66→  const setProvider = useApp((s) => s.setProvider);
    67→  const clearProvider = useApp((s) => s.clearProvider);
    68→
    69→  const [provider, setProviderInfo] = useState<ProviderInfo | null>(null);
    70→  const [activeJobs, setActiveJobs] = useState<Job[]>([]);
    71→  const [pastJobs, setPastJobs] = useState<Job[]>([]);
    72→  const [customRequests, setCustomRequests] = useState<Job[]>([]);
    73→  const [incomingJob, setIncomingJob] = useState<Job | null>(null);
    74→  const [view, setLocalView] = useState<"home" | "job" | "custom" | "new" | "history" | "onboard" | "settings">("home");
    75→  const [selectedJob, setSelectedJob] = useState<Job | null>(null);
    76→
    77→  // Login form state
    78→  const [loginPhone, setLoginPhone] = useState("");
    79→  const [loginPin, setLoginPin] = useState("");
    80→  const [loginError, setLoginError] = useState<string | null>(null);
    81→  const [loginLoading, setLoginLoading] = useState(false);
    82→
    83→  // ── If logged in, fetch provider info ─────────────────
    84→  const refresh = useCallback(async () => {
    85→    if (!providerId) return;
    86→    const res = await fetch(`/api/providers/${providerId}?jobs=true`);
    87→    if (!res.ok) {
    88→      clearProvider();
    89→      return;
    90→    }
    91→    const data = await res.json();
    92→    // Map tenant relation fields to flat fields expected by the provider state
    93→    const p = data.provider;
    94→    const tenantData = p.tenant || {};
    95→    setProviderInfo({
    96→      ...p,
    97→      tenantSlug: tenantData.slug || p.tenantSlug,
    98→      tenantName: tenantData.name || p.tenantName,
    99→      tenantAccent: tenantData.accentColor || p.tenantAccent,
   100→      cityName: p.city?.name || p.cityName,
   101→    });
   102→    setActiveJobs(data.activeJobs);
   103→    setPastJobs(data.pastJobs);
   104→    setCustomRequests(data.customRequests);
   105→  }, [providerId, clearProvider]);
   106→
   107→  useEffect(() => {
   108→    if (providerId) {
   109→      refresh();
   110→      const id = setInterval(refresh, 4000);
   111→      return () => clearInterval(id);
   112→    }
   113→  }, [providerId, refresh]);
   114→
   115→  // ── Poll for incoming broadcast jobs (uses dedicated /incoming endpoint) ──
   116→  useEffect(() => {
   117→    if (!provider || !provider.isOnline) return;
   118→    let cancelled = false;
   119→    async function checkIncoming() {
   120→      if (cancelled || !provider) return;
   121→      try {
   122→        // Use the dedicated /incoming endpoint — queries OrderBroadcast for this provider only
   123→        const res = await fetch(`/api/providers/${provider.id}/incoming`);
   124→        if (!res.ok) return;
   125→        const data = await res.json();
   126→        const myBroadcasts = (data.jobs || []) as Job[];
   127→        if (myBroadcasts.length > 0 && !incomingJob) {
   128→          setIncomingJob(myBroadcasts[0]);
   129→        }
   130→      } catch {
   131→        // silent
   132→      }
   133→    }
   134→    checkIncoming();
   135→    const id = setInterval(checkIncoming, 5000);
   136→    return () => { cancelled = true; clearInterval(id); };
   137→  }, [provider, incomingJob]);
   138→
   139→  // ── Login ─────────────────────────────────────────────
   140→  async function handleLogin() {
   141→    setLoginLoading(true);
   142→    setLoginError(null);
   143→    try {
   144→      const res = await fetch("/api/providers/login", {
   145→        method: "POST",
   146→        headers: { "Content-Type": "application/json" },
   147→        body: JSON.stringify({ tenantSlug: providerTenantSlug || "shanti", phone: loginPhone, pin: loginPin }),
   148→      });
   149→      const data = await res.json();
   150→      if (!res.ok) {
   151→        setLoginError(data.message || data.error || "Login failed");
   152→        return;
   153→      }
   154→      setProvider(data.provider.id, data.provider.tenantSlug);
   155→      setProviderInfo(data.provider);
   156→      setLocalView("onboard");
   157→    } finally {
   158→      setLoginLoading(false);
   159→    }
   160→  }
   161→
   162→  async function toggleOnline() {
   163→    if (!provider) return;
   164→    const newOnline = !provider.isOnline;
   165→    await fetch("/api/providers", {
   166→      method: "PATCH",
   167→      headers: { "Content-Type": "application/json" },
   168→      body: JSON.stringify({ id: provider.id, isOnline: newOnline }),
   169→    });
   170→    setProviderInfo({ ...provider, isOnline: newOnline });
   171→    if (newOnline) toast.success("You're now online — ready to receive jobs");
   172→    else toast.info("You're now offline");
   173→  }
   174→
   175→  async function handleAccept() {
   176→    if (!incomingJob || !provider) return;
   177→    const res = await fetch(`/api/orders/${incomingJob.id}/accept`, {
   178→      method: "POST",
   179→      headers: { "Content-Type": "application/json" },
   180→      body: JSON.stringify({ providerId: provider.id }),
   181→    });
   182→    const data = await res.json();
   183→    if (res.status === 409) {
   184→      toast.error("Already taken by another provider");
   185→      setIncomingJob(null);
   186→      return;
   187→    }
   188→    toast.success(`✅ Order #${incomingJob.code} accepted!`);
   189→    setIncomingJob(null);
   190→    refresh();
   191→  }
   192→
   193→  function handleReject() {
   194→    if (!incomingJob || !provider) return;
   195→    fetch(`/api/orders/${incomingJob.id}/reject`, {
   196→      method: "POST",
   197→      headers: { "Content-Type": "application/json" },
   198→      body: JSON.stringify({ providerId: provider.id }),
   199→    });
   200→    toast.info("Rejected — waiting for next job");
   201→    setIncomingJob(null);
   202→  }
   203→
   204→  async function updateJobStatus(job: Job, status: "picked" | "delivered") {
   205→    await fetch(`/api/orders/${job.id}/status`, {
   206→      method: "POST",
   207→      headers: { "Content-Type": "application/json" },
   208→      body: JSON.stringify({ status, providerId: provider?.id }),
   209→    });
   210→    toast.success(status === "picked" ? "📦 Marked as picked up" : "✅ Marked as delivered");
   211→    refresh();
   212→    setSelectedJob(null);
   213→    setLocalView("home");
   214→  }
   215→
   216→  // ── Render: login screen ──────────────────────────────
   217→  if (!providerId || !provider) {
   218→    return (
   219→      <div className="min-h-screen bg-background flex items-center justify-center p-4">
   220→        <div className="w-full max-w-sm">
   221→          <button
   222→            onClick={() => setView("home")}
   223→            className="text-muted-foreground hover:text-foreground text-sm mb-6 flex items-center gap-1"
   224→          >
   225→            <ArrowLeft className="w-4 h-4" /> Back
   226→          </button>
   227→
   228→          <div className="text-center mb-8">
   229→            <div className="w-16 h-16 rounded-2xl bg-emerald-500/15 border border-emerald-500/20 flex items-center justify-center mx-auto mb-3">
   230→              <Package className="w-8 h-8 text-emerald-400" />
   231→            </div>
   232→            <h1 className="text-xl font-semibold">Provider Login</h1>
   233→            <p className="text-xs text-muted-foreground mt-1">Enter your phone and 4-digit PIN</p>
   234→          </div>
   235→
   236→          <div className="space-y-3">
   237→            <div>
   238→              <label className="text-[11px] uppercase tracking-wider text-muted-foreground">Phone</label>
   239→              <input
   240→                value={loginPhone}
   241→                onChange={(e) => setLoginPhone(e.target.value)}
   242→                className="w-full mt-1 bg-card border border-border rounded-xl px-3 py-2.5 text-sm outline-none focus:border-emerald-500/40"
   243→                placeholder="+91..."
   244→              />
   245→            </div>
   246→            <div>
   247→              <label className="text-[11px] uppercase tracking-wider text-muted-foreground">4-digit PIN</label>
   248→              <input
   249→                value={loginPin}
   250→                onChange={(e) => setLoginPin(e.target.value)}
   251→                maxLength={4}
   252→                inputMode="numeric"
   253→                className="w-full mt-1 bg-card border border-border rounded-xl px-3 py-2.5 text-sm tracking-[0.5em] outline-none focus:border-emerald-500/40 font-mono"
   254→                placeholder="••••"
   255→              />
   256→            </div>
   257→            {loginError && (
   258→              <div className="text-xs text-rose-400 bg-rose-500/10 border border-rose-500/20 rounded-lg p-2">
   259→                {loginError}
   260→              </div>
   261→            )}
   262→            <button
   263→              onClick={handleLogin}
   264→              disabled={loginLoading || !loginPhone || !loginPin}
   265→              className="w-full bg-emerald-500 hover:bg-emerald-400 disabled:opacity-50 text-zinc-950 font-medium py-2.5 rounded-xl text-sm transition-colors"
   266→            >
   267→              {loginLoading ? "Signing in…" : "Sign in"}
   268→            </button>
   269→          </div>
   270→
   271→          <div className="mt-6 p-3 rounded-xl bg-card/60 border border-border/60 text-xs text-muted-foreground">
   272→            <p className="font-medium text-foreground mb-1">Demo providers</p>
   273→            <p>+919811100001 · +919811100002 · +919822200001</p>
   274→            <p className="mt-1">PIN: <span className="font-mono">1234</span></p>
   275→          </div>
   276→        </div>
   277→      </div>
   278→    );
   279→  }
   280→
   281→  // ── Render: onboarding ────────────────────────────────
   282→  if (view === "onboard") {
   283→    return <ProviderOnboarding provider={provider} onDone={() => setLocalView("home")} />;
   284→  }
   285→
   286→  // ── Render: incoming call screen (full-screen takeover) ─
   287→  if (incomingJob) {
   288→    return <IncomingCallScreen job={incomingJob} onAccept={handleAccept} onReject={handleReject} />;
   289→  }
   290→
   291→  // ── Render: job detail ────────────────────────────────
   292→  if (view === "job" && selectedJob) {
   293→    return (
   294→      <JobDetail
   295→        job={selectedJob}
   296→        provider={provider}
   297→        onBack={() => { setLocalView("home"); setSelectedJob(null); }}
   298→        onUpdate={(s) => updateJobStatus(selectedJob, s)}
   299→        refresh={refresh}
   300→      />
   301→    );
   302→  }
   303→
   304→  // ── Render: new manual job ────────────────────────────
   305→  if (view === "new") {
   306→    return (
   307→      <NewJobScreen
   308→        provider={provider}
   309→        onBack={() => setLocalView("home")}
   310→        onCreated={() => { setLocalView("home"); refresh(); }}
   311→      />
   312→    );
   313→  }
   314→
   315→  if (view === "history") {
   316→    return <HistoryScreen jobs={pastJobs} onBack={() => setLocalView("home")} />;
   317→  }
   318→
   319→  if (view === "settings") {
   320→    return <ProviderSettings provider={provider} onBack={() => setLocalView("home")} />;
   321→  }
   322→
   323→  if (view === "custom") {
   324→    return <CustomRequestsScreen jobs={customRequests} provider={provider} onBack={() => setLocalView("home")} onSent={() => { setLocalView("home"); refresh(); }} />;
   325→  }
   326→
   327→  // ── Render: home ──────────────────────────────────────
   328→  return (
   329→    <div className="min-h-screen bg-background pb-20">
   330→      {/* Header */}
   331→      <header className="border-b border-border/60 bg-card/40 backdrop-blur-xl sticky top-0 z-10">
   332→        <div className="px-4 py-3 flex items-center justify-between">
   333→          <button onClick={() => setView("home")} className="text-muted-foreground hover:text-foreground">
   334→            <ArrowLeft className="w-5 h-5" />
   335→          </button>
   336→          <div className="text-center">
   337→            <p className="text-[11px] text-muted-foreground leading-none">{provider.tenantName}</p>
   338→            <p className="text-xs font-medium mt-0.5">{provider.cityName} · {provider.zone}</p>
   339→          </div>
   340→          <button onClick={() => clearProvider()} className="text-muted-foreground hover:text-foreground text-xs">
   341→            Logout
   342→          </button>
   343→        </div>
   344→      </header>
   345→
   346→      <div className="px-4 py-5 space-y-5 max-w-md mx-auto">
   347→        {/* Online toggle */}
   348→        <div className={cn(
   349→          "rounded-2xl p-5 border transition-colors",
   350→          provider.isOnline ? "bg-emerald-500/10 border-emerald-500/30" : "bg-card border-border"
   351→        )}>
   352→          <div className="flex items-center justify-between mb-3">
   353→            <div>
   354→              <p className="text-[11px] uppercase tracking-wider text-muted-foreground">Status</p>
   355→              <p className={cn("text-lg font-semibold mt-0.5", provider.isOnline ? "text-emerald-300" : "text-zinc-400")}>
   356→                {provider.isOnline ? "Online" : "Offline"}
   357→              </p>
   358→            </div>
   359→            <button
   360→              onClick={toggleOnline}
   361→              className={cn(
   362→                "relative w-14 h-8 rounded-full transition-colors",
   363→                provider.isOnline ? "bg-emerald-500" : "bg-zinc-700"
   364→              )}
   365→              aria-label="Toggle online"
   366→            >
   367→              <span className={cn(
   368→                "absolute top-1 w-6 h-6 rounded-full bg-white transition-transform",
   369→                provider.isOnline ? "translate-x-7" : "translate-x-1"
   370→              )} />
   371→            </button>
   372→          </div>
   373→          {provider.isOnline && (
   374→            <p className="text-xs text-emerald-300/80 flex items-center gap-1.5">
   375→              <span className="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-live-dot" />
   376→              Listening for jobs in {provider.zone}
   377→            </p>
   378→          )}
   379→        </div>
   380→
   381→        {/* Stats */}
   382→        <div className="grid grid-cols-3 gap-2">
   383→          <StatBox label="Today" value={String(activeJobs.length)} />
   384→          <StatBox label="Total jobs" value={String(provider.jobsDone)} />
   385→          <StatBox label="Earnings" value={formatINR(provider.earnings)} />
   386→        </div>
   387→
   388→        {/* Rating */}
   389→        <div className="flex items-center justify-between p-3 rounded-xl bg-card border border-border">
   390→          <div className="flex items-center gap-2">
   391→            <Star className="w-4 h-4 fill-amber-400 text-amber-400" />
   392→            <span className="text-sm font-medium">{provider.rating.toFixed(1)}</span>
   393→            <span className="text-xs text-muted-foreground">rating</span>
   394→          </div>
   395→          <span className="text-xs text-muted-foreground">Avg accept: {formatDuration(18)}</span>
   396→        </div>
   397→
   398→        {/* Quick actions */}
   399→        <div className="grid grid-cols-4 gap-2">
   400→          <QuickAction icon={Plus} label="New job" onClick={() => setLocalView("new")} />
   401→          <QuickAction icon={Bell} label="Requests" badge={customRequests.length} onClick={() => setLocalView("custom")} />
   402→          <QuickAction icon={History} label="History" onClick={() => setLocalView("history")} />
   403→          <QuickAction icon={Settings} label="Settings" onClick={() => setLocalView("settings")} />
   404→        </div>
   405→
   406→        {/* Active jobs */}
   407→        <div>
   408→          <h3 className="text-[11px] uppercase tracking-wider text-muted-foreground mb-2">Active jobs</h3>
   409→          {activeJobs.length === 0 ? (
   410→            <EmptyState icon={Package} title="No active jobs" desc="New jobs will appear here" />
   411→          ) : (
   412→            <div className="space-y-2">
   413→              {activeJobs.map((job) => (
   414→                <JobCard key={job.id} job={job} onClick={() => { setSelectedJob(job); setLocalView("job"); }} />
   415→              ))}
   416→            </div>
   417→          )}
   418→        </div>
   419→      </div>
   420→    </div>
   421→  );
   422→}

   438→  return (
   439→    <button
   440→      onClick={onClick}

   442→    >
   443→      <Icon className="w-5 h-5 text-muted-foreground" />
   444→      <span className="text-[11px] text-muted-foreground">{label}</span>
   445→      {badge && badge > 0 && (
   446→        <span className="absolute -top-1 -right-1 w-5 h-5 rounded-full bg-amber-500 text-zinc-950 text-[10px] font-semibold flex items-center justify-center">
   447→          {badge}
   448→        </span>
   449→      )}
   450→    </button>
   451→  );

   464→function JobCard({ job, onClick }: { job: Job; onClick: () => void }) {
   465→  const items = safeParse<OrderItem[]>(job.items, []);
   466→  const labels = job.service ? safeParse<Record<string, string>>(job.service.labels, {}) : {};
   467→  const svcName = job.service ? `${job.service.icon} ${labels.en || job.service.key}` : "Order";
   468→  return (
   469→    <button
   470→      onClick={onClick}
   471→      className="w-full text-left p-3 rounded-xl bg-card border border-border hover:border-emerald-500/30 transition-colors"
   472→    >
   473→      <div className="flex items-center justify-between mb-1">
   474→        <span className="text-xs font-medium">#{job.code}</span>
   475→        <span className={cn("text-[10px] px-1.5 py-0.5 rounded-full", ORDER_STATUS[job.status as keyof typeof ORDER_STATUS]?.bg, ORDER_STATUS[job.status as keyof typeof ORDER_STATUS]?.color)}>
   476→          {ORDER_STATUS[job.status as keyof typeof ORDER_STATUS]?.label || job.status}
   477→        </span>
   478→      </div>
   479→      <p className="text-sm font-medium mb-1">{svcName}</p>
   480→      {items.length > 0 && <p className="text-xs text-muted-foreground">{items.length} item(s) · {items[0]?.name}…</p>}
   481→      {job.description && <p className="text-xs text-muted-foreground truncate">{job.description}</p>}
   482→      <div className="flex items-center justify-between mt-2">
   483→        <span className="text-[11px] text-muted-foreground flex items-center gap-1">
   484→          <MapPin className="w-3 h-3" /> {job.addressArea || "—"}
   485→        </span>
   486→        <ChevronRight className="w-4 h-4 text-muted-foreground" />
   487→      </div>
   488→    </button>
   489→  );
   490→}

   492→function IncomingCallScreen({ job, onAccept, onReject }: { job: Job; onAccept: () => void; onReject: () => void }) {
   493→  const [seconds, setSeconds] = useState(45);
   494→  const items = safeParse<OrderItem[]>(job.items, []);
   495→  const labels = job.service ? safeParse<Record<string, string>>(job.service.labels, {}) : {};
   496→  const svcName = job.service ? `${job.service.icon} ${labels.en || job.service.key}` : "New Job";
   497→
   498→  useEffect(() => {
   499→    const id = setInterval(() => {
   500→      setSeconds((s) => {
   501→        if (s <= 1) {
   502→          clearInterval(id);
   503→          onReject();
   504→          return 0;
   505→        }

   900→function ChargesSection({ job, provider, onChargesSet }: { job: Job; provider: ProviderInfo | null; onChargesSet: (updated: Partial<Job>) => void }) {
   901→  const [modalOpen, setModalOpen] = useState(false);
   902→  const [deliveryCharge, setDeliveryCharge] = useState("");
   903→  const [serviceCharge, setServiceCharge] = useState("");
   904→  const [addonsCharge, setAddonsCharge] = useState("");
   905→  const [itemsTotal, setItemsTotal] = useState("");
   906→  const [loading, setLoading] = useState(false);
   907→
   908→  const isOrder = job.kind === "order";
   909→  const isBook = job.kind === "book";
   910→
   911→  // Pre-fill with service defaults when modal opens
   912→  useEffect(() => {
   913→    if (modalOpen) {
   914→      if (isOrder) {
   915→        setDeliveryCharge(job.deliveryCharge ? String(job.deliveryCharge / 100) : (job.service?.defaultDeliveryCharge ? String(job.service.defaultDeliveryCharge / 100) : ""));
   916→        setItemsTotal(job.itemsTotal ? String(job.itemsTotal / 100) : "");
   917→      }
   918→      if (isBook) {
   919→        setServiceCharge(job.serviceCharge ? String(job.serviceCharge / 100) : (job.service?.defaultServiceCharge ? String(job.service.defaultServiceCharge / 100) : ""));
   920→        setAddonsCharge(job.addonsCharge ? String(job.addonsCharge / 100) : "");
   921→      }
   922→    }
   923→  }, [modalOpen]);
   924→
   925→  const totalRupees = job.totalAmount ? (job.totalAmount / 100) : 0;
   926→
   927→  async function sendCharges() {
   928→    if (!provider) return;
   929→    setLoading(true);
   930→    try {
   931→      const body: Record<string, unknown> = { providerId: provider.id, tenantSlug: provider.tenantSlug };
   932→      if (isOrder) {
   933→        body.deliveryCharge = deliveryCharge || 0;
   934→        body.itemsTotal = itemsTotal || 0;
   935→      }
   936→      if (isBook) {
   937→        body.serviceCharge = serviceCharge || 0;
   938→        body.addonsCharge = addonsCharge || 0;
   939→      }
   940→      const res = await fetch(`/api/orders/${job.id}/charges`, {
   941→        method: "POST",
   942→        headers: { "Content-Type": "application/json" },
   943→        body: JSON.stringify(body),
   944→      });
   945→      const data = await res.json();
   946→      if (res.ok) {
   947→        toast.success("📋 Charges sent to customer — waiting for agreement");
   948→        setModalOpen(false);
   949→        // Update local job with the new charges
   950→        if (data.charges) {
   951→          onChargesSet({
   952→            deliveryCharge: data.charges.deliveryCharge,
   953→            serviceCharge: data.charges.serviceCharge,
   954→            addonsCharge: data.charges.addonsCharge,
   955→            itemsTotal: data.charges.itemsTotal,
   956→            totalAmount: data.charges.totalAmount,
   957→            chargesConfirmed: data.charges.chargesConfirmed,
   958→          });
   959→        }
   960→      } else {
   961→        toast.error(data.error || "Failed to set charges");
   962→      }
   963→    } finally {
   964→      setLoading(false);
   965→    }

  1473→function ProviderSettings({ provider, onBack }: { provider: ProviderInfo; onBack: () => void }) {
  1474→  const [upiIds, setUpiIds] = useState<Array<{ id: string; vpa: string; label: string; isDefault: boolean }>>([]);
  1475→  const [googleReviewUrl, setGoogleReviewUrl] = useState("");
  1476→  const [feedbackEnabled, setFeedbackEnabled] = useState(true);
  1477→  const [loading, setLoading] = useState(false);
  1478→  const [newVpa, setNewVpa] = useState("");
  1479→  const [newLabel, setNewLabel] = useState("");
  1480→
  1481→  useEffect(() => {
  1482→    fetch(`/api/providers/${provider.id}/upi-ids`).then((r) => r.json()).then((d) => setUpiIds(d.upiIds || []));
  1483→    fetch(`/api/providers/${provider.id}/feedback`).then((r) => r.json()).then((d) => {
  1484→      setGoogleReviewUrl(d.googleReviewUrl || "");
  1485→      setFeedbackEnabled(d.feedbackEnabled ?? true);
  1486→    });
  1487→  }, [provider.id]);
  1488→
  1489→  async function addUpi() {
  1490→    if (!newVpa || !/^[a-zA-Z0-9.\-_]+@[a-zA-Z0-9.\-_]+$/.test(newVpa)) {
  1491→      toast.error("Invalid UPI ID format (e.g. name@bank)");
  1492→      return;
  1493→    }
  1494→    const newUpis = [...upiIds, {
  1495→      id: Math.random().toString(36).slice(2),
  1496→      vpa: newVpa,
  1497→      label: newLabel || newVpa.split("@")[0],
  1498→      isDefault: upiIds.length === 0, // first one is default
  1499→    }];
  1500→    await saveUpis(newUpis);
  1501→    setNewVpa("");
  1502→    setNewLabel("");
  1503→  }
  1504→
  1505→  async function removeUpi(id: string) {
  1506→    const filtered = upiIds.filter((u) => u.id !== id);
  1507→    // If we removed the default, make the first one default
  1508→    if (!filtered.find((u) => u.isDefault) && filtered.length > 0) {
  1509→      filtered[0].isDefault = true;
  1510→    }
  1511→    await saveUpis(filtered);
  1512→  }
  1513→
  1514→  async function setDefault(id: string) {
  1515→    const updated = upiIds.map((u) => ({ ...u, isDefault: u.id === id }));
  1516→    await saveUpis(updated);
  1517→  }
  1518→
  1519→  async function saveUpis(newUpis: Array<{ id: string; vpa: string; label: string; isDefault: boolean }>) {
  1520→    setLoading(true);
  1521→    try {
  1522→      const res = await fetch(`/api/providers/${provider.id}/upi-ids`, {
  1523→        method: "PATCH",
  1524→        headers: { "Content-Type": "application/json" },
  1525→        body: JSON.stringify({ upiIds: newUpis }),
  1526→      });
  1527→      if (res.ok) {
  1528→        setUpiIds(newUpis);
  1529→        toast.success("UPI IDs updated");
  1530→      }
  1531→    } finally {
  1532→      setLoading(false);
  1533→    }
  1534→  }
  1535→
  1536→  async function saveFeedback() {
  1537→    setLoading(true);
  1538→    try {
  1539→      await fetch(`/api/providers/${provider.id}/feedback`, {
  1540→        method: "PATCH",
  1541→        headers: { "Content-Type": "application/json" },
  1542→        body: JSON.stringify({ googleReviewUrl, feedbackEnabled }),
  1543→      });
  1544→      toast.success("Feedback settings saved");
  1545→    } finally {
  1546→      setLoading(false);
  1547→    }
  1548→  }
  1549→
  1550→  return (
  1551→    <div className="min-h-screen bg-background">
  1552→      <header className="border-b border-border/60 bg-card/40 backdrop-blur-xl sticky top-0 z-10">
  1553→        <div className="px-4 py-3 flex items-center gap-3">
  1554→          <button onClick={onBack} className="text-muted-foreground hover:text-foreground">
  1555→            <ArrowLeft className="w-5 h-5" />
  1556→          </button>
  1557→          <h2 className="text-sm font-medium">Settings</h2>
  1558→        </div>
  1559→      </header>
  1560→
  1561→      <div className="px-4 py-4 space-y-5 max-w-md mx-auto">
  1562→        {/* UPI IDs */}
  1563→        <div className="p-4 rounded-xl border border-border bg-card">
  1564→          <p className="text-sm font-medium mb-1">UPI IDs</p>
  1565→          <p className="text-xs text-muted-foreground mb-3">Add multiple UPI IDs. The default one is used for payment links.</p>
  1566→
  1567→          {upiIds.length > 0 && (
  1568→            <div className="space-y-1.5 mb-3">
  1569→              {upiIds.map((upi) => (
  1570→                <div key={upi.id} className={cn(
  1571→                  "p-2.5 rounded-lg border flex items-center gap-2",
  1572→                  upi.isDefault ? "border-emerald-500/30 bg-emerald-500/5" : "border-border"
  1573→                )}>
  1574→                  <div className="flex-1 min-w-0">
  1575→                    <p className="text-sm font-mono truncate">{upi.vpa}</p>
  1576→                    <p className="text-[10px] text-muted-foreground">{upi.label}</p>
  1577→                  </div>
  1578→                  {upi.isDefault ? (
  1579→                    <span className="text-[10px] px-1.5 py-0.5 rounded bg-emerald-500/15 text-emerald-300">Default</span>
  1580→                  ) : (
  1581→                    <button onClick={() => setDefault(upi.id)} className="text-[10px] px-1.5 py-0.5 rounded border border-border text-muted-foreground hover:text-foreground">
  1582→                      Set default
  1583→                    </button>
  1584→                  )}
  1585→                  <button onClick={() => removeUpi(upi.id)} className="text-rose-400 hover:text-rose-300">
  1586→                    <X className="w-3.5 h-3.5" />
  1587→                  </button>
  1588→                </div>
  1589→              ))}
  1590→            </div>
  1591→          )}
  1592→
  1593→          {/* Add new UPI */}
  1594→          <div className="space-y-2 pt-2 border-t border-border/40">
  1595→            <input
  1596→              value={newVpa}
  1597→              onChange={(e) => setNewVpa(e.target.value)}
  1598→              placeholder="yourname@okhdfcbank"
  1599→              className="w-full bg-background border border-border rounded-lg px-3 py-2 text-sm font-mono outline-none focus:border-emerald-500/40"
  1600→            />
  1601→            <input
  1602→              value={newLabel}
  1603→              onChange={(e) => setNewLabel(e.target.value)}
  1604→              placeholder="Label (e.g. Personal, Business)"
  1605→              className="w-full bg-background border border-border rounded-lg px-3 py-2 text-sm outline-none focus:border-emerald-500/40"
  1606→            />
  1607→            <button
  1608→              onClick={addUpi}
  1609→              disabled={!newVpa || loading}
  1610→              className="w-full bg-emerald-500/15 hover:bg-emerald-500/25 border border-emerald-500/30 text-emerald-300 text-xs font-medium py-2 rounded-lg"
  1611→            >
  1612→              + Add UPI ID
  1613→            </button>
  1614→          </div>
  1615→        </div>
  1616→
  1617→        {/* Google review URL */}
  1618→        <div className="p-4 rounded-xl border border-border bg-card">
  1619→          <p className="text-sm font-medium mb-1">Google Business review</p>
  1620→          <p className="text-xs text-muted-foreground mb-3">After delivery, customers get a link to review you on Google. Get this from your Google Business profile.</p>
  1621→          <input
  1622→            value={googleReviewUrl}
  1623→            onChange={(e) => setGoogleReviewUrl(e.target.value)}
  1624→            placeholder="https://g.page/r/CXXXX/review"
  1625→            className="w-full bg-background border border-border rounded-lg px-3 py-2 text-sm font-mono outline-none focus:border-emerald-500/40"
  1626→          />
  1627→          <p className="text-[10px] text-muted-foreground mt-1">
  1628→            Tip: Go to Google Business Profile → Ask for reviews → copy the link.
  1629→          </p>
  1630→        </div>
  1631→
  1632→        {/* Feedback toggle */}
  1633→        <div className="p-4 rounded-xl border border-border bg-card">
  1634→          <div className="flex items-center justify-between">
  1635→            <div>
  1636→              <p className="text-sm font-medium">Ask customers for feedback</p>
  1637→              <p className="text-xs text-muted-foreground">After delivery, ask for a 1-5★ rating on WhatsApp</p>
  1638→            </div>
  1639→            <button
  1640→              onClick={() => setFeedbackEnabled(!feedbackEnabled)}
  1641→              className={cn("relative w-10 h-6 rounded-full transition-colors", feedbackEnabled ? "bg-emerald-500" : "bg-zinc-700")}
  1642→            >
  1643→              <span className={cn("absolute top-1 w-4 h-4 rounded-full bg-white transition-transform", feedbackEnabled ? "translate-x-5" : "translate-x-1")} />
  1644→            </button>
  1645→          </div>
  1646→        </div>
  1647→
  1648→        <button
  1649→          onClick={saveFeedback}
  1650→          disabled={loading}
  1651→          className="w-full bg-emerald-500 hover:bg-emerald-400 disabled:opacity-50 text-zinc-950 font-medium py-2.5 rounded-xl text-sm"
  1652→        >
  1653→          {loading ? "Saving…" : "Save feedback settings"}
  1654→        </button>
  1655→      </div>
  1656→    </div>
  1657→  );
  1658→}
